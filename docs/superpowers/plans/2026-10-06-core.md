# Core Library Implementation Plan

**Goal:** Build `@sharepay/core`. It holds the workspace scaffold, the shared DTOs, the token registry,
bigint money math, a signed Binance Web3 API client with DevEx telemetry, typed RWA/Trading/Transaction/Wallet
wrappers, the x402 v2 codec, EIP-3009 helpers, and a BSC chain reader with a simulate → sign → broadcast
transaction pipeline.

**Architecture:** This is an npm-workspaces monorepo. Each workspace package exports its TypeScript sources
directly: `"."` maps to `./src/index.ts` and `"./*"` maps to `./src/*.ts`, so there is no build step and
servers run with `tsx`. The core package has no workspace dependencies. Every external response is
validated with permissive zod schemas, and every signing function takes an injected signer so that tests
use throwaway keys.

**Tech Stack:** Node.js >= 22.12, TypeScript ~5.9.3 (strict, NodeNext), viem ^2.57.3, zod ^3.25.76,
vitest ^3.2.7, tsx ^4.23.15

**Source spec:** docs/superpowers/specs/2026-10-06-sharepay-design.md

**Conventions for every task in this plan:**
- Relative imports use the `.js` extension (`import { x } from "./money.js";`).
- Tests import from `vitest` explicitly.
- No test touches the network or a real secret.
- The gate script `npm run check -- <test file>` runs `tsc -p tsconfig.json` (whole repo) and then the
  named vitest file. Created in Task 1.

import { formatUnits, parseUnits } from "viem";

export const ONE: bigint = 10n ** 18n;

export function toUnits(value: string, decimals: number): bigint {
  const trimmed = value.trim();
  if (!/^\d+(\.\d+)?$/.test(trimmed)) {
    throw new Error(`invalid decimal: ${value}`);
  }
  return parseUnits(trimmed, decimals);
}

export function ceilDiv(a: bigint, b: bigint): bigint {
  if (b === 0n) {
    throw new Error("division by zero");
  }
  return (a + b - 1n) / b;
}

export function mulDivUp(a: bigint, b: bigint, c: bigint): bigint {
  return ceilDiv(a * b, c);
}

export function mulDivDown(a: bigint, b: bigint, c: bigint): bigint {
  if (c === 0n) {
    throw new Error("division by zero");
  }
  return (a * b) / c;
}

export function addBps(amount: bigint, bps: number): bigint {
  return mulDivUp(amount, BigInt(10_000 + bps), 10_000n);
}

export function formatAmount(amount: bigint, decimals: number, maxFractionDigits: number): string {
  const full = formatUnits(amount, decimals);
  const [intPart = "0", fraction = ""] = full.split(".");
  const cut = fraction.slice(0, maxFractionDigits).replace(/0+$/, "");
  return cut.length > 0 ? `${intPart}.${cut}` : intPart;
}

export function formatSignificant(amount: bigint, decimals: number, significant: number): string {
  if (amount === 0n) {
    return "0";
  }
  const full = formatUnits(amount, decimals);
  const [intPart = "0", fraction = ""] = full.split(".");
  if (intPart !== "0") {
    const cut = fraction.slice(0, 4).replace(/0+$/, "");
    return cut.length > 0 ? `${intPart}.${cut}` : intPart;
  }
  const firstNonZero = fraction.search(/[1-9]/);
  if (firstNonZero === -1) {
    return "0";
  }
  const cut = fraction.slice(0, firstNonZero + significant).replace(/0+$/, "");
  return `0.${cut}`;
}

export function sharesFromTokens(tokens: bigint, sharesPerToken: bigint): bigint {
  return mulDivDown(tokens, sharesPerToken, ONE);
}

export function usdValue(tokens: bigint, priceUsd: string): string {
  const price = toUnits(priceUsd, 18);
  return formatAmount(mulDivDown(tokens, price, ONE), 18, 2);
}

import { createHmac } from "node:crypto";

export type HttpMethod = "GET" | "POST";

export function signPayload(input: {
  timestamp: string;
  method: HttpMethod;
  requestPath: string;
  body: string;
  secretKey: string;
}): string {
  const preHash = `${input.timestamp}${input.method}${input.requestPath}${input.body}`;
  return createHmac("sha256", input.secretKey).update(preHash, "utf8").digest("base64");
}

export function encodeQuery(query: Record<string, string | undefined>): string {
  const parts: string[] = [];
  for (const [key, value] of Object.entries(query)) {
    if (value === undefined) continue;
    const encodedValue = encodeURIComponent(value).replace(/%2C/gi, ",");
    parts.push(`${encodeURIComponent(key)}=${encodedValue}`);
  }
  return parts.join("&");
}

export function parseJsonl(text: string): ApiCallEvent[] {

    .split(/\r?\n/)
    .filter((line) => line.trim().length > 0)
    .map((line) => JSON.parse(line) as ApiCallEvent);

function percentile(sorted: number[], p: number): number {
  if (sorted.length === 0) return 0;
  const rank = Math.ceil((p / 100) * sorted.length);
  const index = Math.min(Math.max(rank - 1, 0), sorted.length - 1);
  return sorted[index] ?? 0;
}

export function summarizeEvents(events: ApiCallEvent[]): EndpointSummary[] {
  const byPath = new Map<string, ApiCallEvent[]>();
  for (const event of events) {
    const list = byPath.get(event.path) ?? [];
    list.push(event);
    byPath.set(event.path, list);
  }
  return [...byPath.keys()].sort().map((path) => {
    const list = byPath.get(path) ?? [];
    const latencies = list.map((e) => e.latencyMs).sort((a, b) => a - b);
    const codes = new Map<string, { code: string; count: number; msg: string }>();
    let errors = 0;
    for (const e of list) {
      const isError = e.httpStatus >= 400 || e.httpStatus === 0 || (e.code !== null && e.code !== 0);
      if (isError) errors += 1;
      const key = e.code !== null ? String(e.code) : `http ${e.httpStatus}`;
      const entry = codes.get(key) ?? { code: key, count: 0, msg: e.msg ?? "" };
      entry.count += 1;
      codes.set(key, entry);
    }
    return {
      path,
      calls: list.length,
      errors,
      p50Ms: percentile(latencies, 50),
      p95Ms: percentile(latencies, 95),
      codes: [...codes.values()],
    };

import { z } from "zod";
import type { TelemetrySink } from "../devex.js";
import { encodeQuery, signPayload, type HttpMethod } from "./sign.js";

export const DEFAULT_BASE_URL = "https://web3.binance.com";
const BUILD_PREFIX = "/build";

export class BinanceApiError extends Error {
  readonly path: string;
  readonly httpStatus: number;
  readonly code: number | null;
  readonly apiMsg: string;

  constructor(path: string, httpStatus: number, code: number | null, apiMsg: string) {
    super(`Binance API ${path} failed: ${code ?? `http ${httpStatus}`} ${apiMsg}`);
    this.name = "BinanceApiError";
    this.path = path;
    this.httpStatus = httpStatus;
    this.code = code;
    this.apiMsg = apiMsg;
  }
}

export interface BinanceClientOptions {
  credentials: { apiKey: string; secretKey: string };
  baseUrl?: string;
  fetchImpl?: typeof fetch;
  now?: () => Date;
  sleep?: (ms: number) => Promise<void>;
  timeoutMs?: number;
  minIntervalMs?: number;
  telemetry?: TelemetrySink;
}

export interface BinanceClient {
  get<T>(path: string, query: Record<string, string | undefined>, schema: z.ZodType<T, z.ZodTypeDef, unknown>): Promise<T>;
  post<T>(path: string, body: unknown, schema: z.ZodType<T, z.ZodTypeDef, unknown>): Promise<T>;
}

const EnvelopeSchema = z
  .object({
    code: z.union([z.number(), z.string()]).nullable().optional(),
    msg: z.string().nullable().optional(),
    data: z.unknown(),
  })
  .passthrough();

function errorMessage(error: unknown): string {
  return error instanceof Error ? error.message : String(error);
}

export function createBinanceClient(options: BinanceClientOptions): BinanceClient {
  const baseUrl = options.baseUrl ?? DEFAULT_BASE_URL;
  const fetchImpl = options.fetchImpl ?? fetch;
  const now = options.now ?? (() => new Date());
  const sleep = options.sleep ?? ((ms: number) => new Promise<void>((resolve) => setTimeout(resolve, ms)));
  const timeoutMs = options.timeoutMs ?? 15_000;
  const minIntervalMs = options.minIntervalMs ?? 250;
  const telemetry: TelemetrySink = options.telemetry ?? (() => undefined);
  const lastStart = new Map<string, number>();

  async function pace(path: string): Promise<void> {
    const previous = lastStart.get(path);
    if (previous !== undefined) {
      const wait = previous + minIntervalMs - now().getTime();
      if (wait > 0) await sleep(wait);
    }
    lastStart.set(path, now().getTime());
  }

  async function request<T>(
    method: HttpMethod,
    path: string,
    query: Record<string, string | undefined>,
    body: unknown,
    schema: z.ZodType<T, z.ZodTypeDef, unknown>,
  ): Promise<T> {
    const qs = encodeQuery(query);
    const requestPath = `${BUILD_PREFIX}${path}${qs ? `?${qs}` : ""}`;
    const bodyText = method === "POST" ? JSON.stringify(body ?? {}) : "";
    for (let attempt = 0; attempt < 2; attempt += 1) {
      await pace(path);
      const timestamp = now().toISOString();
      const headers: Record<string, string> = {
        "X-OC-APIKEY": options.credentials.apiKey,
        "X-OC-TIMESTAMP": timestamp,
        "X-OC-SIGN": signPayload({ timestamp, method, requestPath, body: bodyText, secretKey: options.credentials.secretKey }),
        "X-OC-RECV-WINDOW": "10000",
        Accept: "application/json",
      };
      if (method === "POST") headers["Content-Type"] = "application/json";
      const started = Date.now();
      let response: Response;
      try {
        response = await fetchImpl(`${baseUrl}${requestPath}`, {
          method,
          headers,
          body: method === "POST" ? bodyText : undefined,
          signal: AbortSignal.timeout(timeoutMs),
        });
      } catch (error) {
        telemetry({ at: timestamp, method, path, httpStatus: 0, code: null, msg: errorMessage(error), latencyMs: Date.now() - started });
        throw new BinanceApiError(path, 0, null, errorMessage(error));
      }
      const latencyMs = Date.now() - started;
      if (response.status === 429 && attempt === 0) {
        telemetry({ at: timestamp, method, path, httpStatus: 429, code: null, msg: "rate limited", latencyMs });
        const retryAfter = Number(response.headers.get("Retry-After") ?? "1");
        const seconds = Number.isFinite(retryAfter) && retryAfter > 0 ? Math.min(retryAfter, 5) : 1;
        await sleep(seconds * 1000);
        continue;
      }
      const text = await response.text();
      let parsedJson: unknown = null;
      try {
        parsedJson = text.length > 0 ? JSON.parse(text) : null;
      } catch {
        parsedJson = null;
      }
      const envelope = EnvelopeSchema.safeParse(parsedJson);
      if (!envelope.success) {
        telemetry({ at: timestamp, method, path, httpStatus: response.status, code: null, msg: `non-envelope body: ${text.slice(0, 200)}`, latencyMs });
        throw new BinanceApiError(path, response.status, null, `unexpected response body: ${text.slice(0, 200)}`);
      }
      const rawCode = envelope.data.code;
      const code = rawCode === null || rawCode === undefined ? null : Number(rawCode);
      const msg = envelope.data.msg ?? "";
      telemetry({ at: timestamp, method, path, httpStatus: response.status, code, msg, latencyMs });
      const ok = response.ok && (code === 0 || code === null);
      if (!ok) {
        throw new BinanceApiError(path, response.status, code, msg);
      }
      const data = schema.safeParse(envelope.data.data);
      if (!data.success) {
        throw new BinanceApiError(path, response.status, code, `schema mismatch: ${data.error.message.slice(0, 300)}`);
      }
      return data.data;
    }
    throw new BinanceApiError(path, 429, null, "rate limited after retry");
  }

  return {
    get: (path, query, schema) => request("GET", path, query, undefined, schema),
    post: (path, body, schema) => request("POST", path, {}, body, schema),
  };
}

function humanize(key: string): string {
  const spaced = key.replace(/([a-z0-9])([A-Z])/g, "$1 $2").toLowerCase();
  return spaced.charAt(0).toUpperCase() + spaced.slice(1);
}

export function protectionLinks(
  protections: Record<string, unknown> | null | undefined,
): Array<{ name: string; url: string }> {
  if (!protections) return [];
  const links: Array<{ name: string; url: string }> = [];
  for (const [key, value] of Object.entries(protections)) {
    if (value === null || typeof value !== "object") continue;
    const entry = value as Record<string, unknown>;
    if (entry.supported === false) continue;
    const url = [entry.url, entry.link, entry.reportUrl].find((v): v is string => typeof v === "string" && v.length > 0);
    if (url) links.push({ name: humanize(key), url });
  }
  return links;
}

export function encodeBase64Json(value: unknown): string {
  return Buffer.from(JSON.stringify(value), "utf8").toString("base64");
}

export function decodeBase64Json(value: string): unknown {
  const text = Buffer.from(value.trim(), "base64").toString("utf8");

function tryDecode(value: string): unknown {
  try {
    return decodeBase64Json(value);
  } catch {
    return undefined;
  }

export function parseChallenge(headerValue: string | null, body: unknown): ParsedChallenge {
  const candidates: unknown[] = [];
  if (headerValue) candidates.push(tryDecode(headerValue));
  if (body !== null && typeof body === "object") {
    const record = body as Record<string, unknown>;
    if (record.paymentRequired !== undefined) candidates.push(record.paymentRequired);
    candidates.push(body);
  }
  for (const candidate of candidates) {
    const parsed = PaymentRequiredSchema.safeParse(candidate);
    if (parsed.success) {
      return {
        required: parsed.data,
        raw: candidate as Record<string, unknown>,
        headerValue: headerValue && tryDecode(headerValue) !== undefined ? headerValue : null,
      };
    }
  }
  throw new X402Error("invalid_challenge", "response carried no valid x402 v2 PaymentRequired object");
}

export function challengeHeaderValue(challenge: ParsedChallenge): string {
  return challenge.headerValue ?? encodeBase64Json(challenge.raw);
}

function transferMethod(accept: PaymentRequirement): string {
  const method = accept.extra?.assetTransferMethod;
  return typeof method === "string" ? method : "eip3009";
}

export function selectEip3009Accept(
  challenge: ParsedChallenge,
  options: { network: string; preferredAssets: string[]; maxAmount: bigint },
): SelectedAccept {
  const rejected: string[] = [];
  for (const preferred of options.preferredAssets) {
    for (const accept of challenge.required.accepts) {
      if (!sameAddress(accept.asset, preferred)) continue;
      const stable = findStableByAddress(accept.asset);
      if (accept.scheme !== "exact" || accept.network !== options.network) continue;
      if (transferMethod(accept) !== "eip3009" || !stable || stable.eip3009 === null) continue;
      let amount: bigint;
      try {
        amount = BigInt(accept.amount);
      } catch {
        rejected.push(`${accept.asset}: bad amount ${accept.amount}`);
        continue;
      }
      if (amount > options.maxAmount) {
        rejected.push(`${stable.symbol}: amount ${accept.amount} above cap ${options.maxAmount}`);
        continue;
      }
      return { accept, stable };
    }
  }
  const offered = challenge.required.accepts
    .map((a) => `${a.network}/${a.asset}/${transferMethod(a)}/${a.amount}`)
    .join("; ");
  throw new X402Error(
    "no_supported_accept",
    `no exact eip3009 U/USD1 accept on ${options.network} within cap (offered: ${offered}${rejected.length ? `; rejected: ${rejected.join("; ")}` : ""})`,
  );
}

export function encodePaymentPayload(payload: PaymentPayload): string {
  return encodeBase64Json(payload);
}

export function decodePaymentPayload(headerValue: string): PaymentPayload {
  const decoded = tryDecode(headerValue);
  const parsed = PaymentPayloadSchema.safeParse(decoded);
  if (!parsed.success) {
    throw new X402Error("invalid_payload", `invalid PAYMENT-SIGNATURE payload: ${parsed.error.message.slice(0, 200)}`);
  }
  return parsed.data;
}

export function encodeSettlementResponse(response: SettlementResponse): string {
  return encodeBase64Json(response);
}

export function decodeSettlementResponse(headerValue: string | null): SettlementResponse | null {
  if (!headerValue) return null;
  const parsed = SettlementResponseSchema.safeParse(tryDecode(headerValue));
  return parsed.success ? parsed.data : null;
}

export function canonicalJson(value: unknown): string {
  if (Array.isArray(value)) {
    return `[${value.map((item) => canonicalJson(item)).join(",")}]`;
  }
  if (value !== null && typeof value === "object") {
    const record = value as Record<string, unknown>;
    const keys = Object.keys(record).filter((key) => record[key] !== undefined).sort();
    return `{${keys.map((key) => `${JSON.stringify(key)}:${canonicalJson(record[key])}`).join(",")}}`;
  }
  return JSON.stringify(value);
}

import { createPublicClient, http, type Address, type Hex, type Transport } from "viem";
import { bsc } from "viem/chains";

export const DEFAULT_BSC_RPC_URL = "https://bsc-dataseed.bnbchain.org";

const CHAIN_ABI = [
  { type: "function", name: "balanceOf", stateMutability: "view", inputs: [{ name: "owner", type: "address" }], outputs: [{ name: "", type: "uint256" }] },
  { type: "function", name: "allowance", stateMutability: "view", inputs: [{ name: "owner", type: "address" }, { name: "spender", type: "address" }], outputs: [{ name: "", type: "uint256" }] },
  { type: "function", name: "authorizationState", stateMutability: "view", inputs: [{ name: "authorizer", type: "address" }, { name: "nonce", type: "bytes32" }], outputs: [{ name: "", type: "bool" }] },
  { type: "function", name: "uiMultiplier", stateMutability: "view", inputs: [], outputs: [{ name: "", type: "uint256" }] },
] as const;

export interface ChainReader {
  balanceOf(token: Address, owner: Address): Promise<bigint>;
  allowance(token: Address, owner: Address, spender: Address): Promise<bigint>;
  authorizationState(token: Address, authorizer: Address, nonce: Hex): Promise<boolean>;
  uiMultiplier(token: Address): Promise<bigint>;
  nativeBalance(address: Address): Promise<bigint>;
  pendingNonce(address: Address): Promise<number>;
  waitForReceipt(hash: Hex, timeoutMs: number): Promise<{ status: "success" | "reverted"; blockNumber: bigint }>;
}

export function createChainReader(options?: { rpcUrl?: string; transport?: Transport; pollMs?: number }): ChainReader {
  const client = createPublicClient({
    chain: bsc,
    transport: options?.transport ?? http(options?.rpcUrl ?? DEFAULT_BSC_RPC_URL),
  });
  const pollMs = options?.pollMs ?? 1000;
  return {
    balanceOf: (token, owner) => client.readContract({ address: token, abi: CHAIN_ABI, functionName: "balanceOf", args: [owner] }),
    allowance: (token, owner, spender) =>
      client.readContract({ address: token, abi: CHAIN_ABI, functionName: "allowance", args: [owner, spender] }),
    authorizationState: (token, authorizer, nonce) =>
      client.readContract({ address: token, abi: CHAIN_ABI, functionName: "authorizationState", args: [authorizer, nonce] }),
    uiMultiplier: (token) => client.readContract({ address: token, abi: CHAIN_ABI, functionName: "uiMultiplier" }),
    nativeBalance: (address) => client.getBalance({ address }),
    async pendingNonce(address) {
      const hex = await client.request({ method: "eth_getTransactionCount", params: [address, "pending"] });
      return Number(BigInt(hex));
    },
    async waitForReceipt(hash, timeoutMs) {
      const deadline = Date.now() + timeoutMs;
      for (;;) {
        const receipt = (await client.request({ method: "eth_getTransactionReceipt", params: [hash] })) as
          | { status?: string | null; blockNumber?: string | null }
          | null;
        if (receipt && receipt.blockNumber) {
          return { status: receipt.status === "0x1" ? "success" : "reverted", blockNumber: BigInt(receipt.blockNumber) };
        }
        if (Date.now() >= deadline) {
          throw new Error(`receipt timeout for ${hash}`);
        }
        await new Promise((resolve) => setTimeout(resolve, pollMs));
      }
    },
  };
}

import type { Address, Hex, LocalAccount } from "viem";
import type { SimulationResult, TransactionApi } from "../binance/transaction.js";
import type { ChainReader } from "./rpc.js";

export interface LegacyTxRequest {
  to: Address;
  data: Hex;
  value: bigint;
  gas: bigint;
  gasPrice: bigint;
  nonce: number;
  chainId: 56;
}

export interface TxSigner {
  address: Address;
  signLegacyTx(tx: LegacyTxRequest): Promise<Hex>;
}

export interface PreparedTx {
  to: Address;
  data: Hex;
  value: bigint;
}

export interface PipelineDeps {
  tx: TransactionApi;
  chain: ChainReader;
  signer: TxSigner;
}

export class TxFailedError extends Error {
  readonly stage: "broadcast" | "receipt";
  readonly txHash: Hex | null;

  constructor(stage: "broadcast" | "receipt", message: string, txHash: Hex | null) {
    super(message);
    this.name = "TxFailedError";
    this.stage = stage;
    this.txHash = txHash;
  }
}

export function simulatePrepared(tx: TransactionApi, from: Address, prepared: PreparedTx): Promise<SimulationResult> {
  return tx.simulate({ from, to: prepared.to, value: prepared.value, data: prepared.data });
}

export async function sendPrepared(
  deps: PipelineDeps,
  prepared: PreparedTx,
  options: { mevProtection: boolean; receiptTimeoutMs: number; gasFallback: bigint },
): Promise<{ txHash: Hex }> {
  const gasPrice = await deps.tx.gasPrice();
  let gas: bigint;
  try {
    const estimate = await deps.tx.gasLimit({ from: deps.signer.address, to: prepared.to, value: prepared.value, data: prepared.data });
    gas = (estimate * 12n) / 10n;
  } catch {
    gas = options.gasFallback;
  }
  const nonce = await deps.chain.pendingNonce(deps.signer.address);
  const raw = await deps.signer.signLegacyTx({
    to: prepared.to,
    data: prepared.data,
    value: prepared.value,
    gas,
    gasPrice,
    nonce,
    chainId: 56,
  });
  let txHash: Hex;
  try {
    const broadcast = await deps.tx.broadcast({
      signedTransaction: raw,
      address: deps.signer.address,
      enableMevProtection: options.mevProtection,
    });
    txHash = broadcast.txHash as Hex;
  } catch (error) {
    throw new TxFailedError("broadcast", error instanceof Error ? error.message : String(error), null);
  }
  const receipt = await deps.chain.waitForReceipt(txHash, options.receiptTimeoutMs);
  if (receipt.status !== "success") {
    throw new TxFailedError("receipt", `transaction ${txHash} reverted`, txHash);
  }
  return { txHash };
}

export type WalletMutex = <T>(fn: () => Promise<T>) => Promise<T>;

export function createWalletMutex(): WalletMutex {
  let tail: Promise<unknown> = Promise.resolve();
  return <T>(fn: () => Promise<T>): Promise<T> => {
    const run = tail.then(fn, fn);
    tail = run.catch(() => undefined);
    return run;
  };
}

export function viemTxSigner(account: LocalAccount): TxSigner {
  return {
    address: account.address,
    signLegacyTx: (tx) =>
      account.signTransaction({
        type: "legacy",
        to: tx.to,
        data: tx.data,
        value: tx.value,
        gas: tx.gas,
        gasPrice: tx.gasPrice,
        nonce: tx.nonce,
        chainId: tx.chainId,
      }),
  };
}
