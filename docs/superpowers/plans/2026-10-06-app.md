# Checkout App Implementation Plan

**Goal:** Ship the hosted "Pay with shares" checkout. It has two halves:
- `@sharepay/checkout`: an Express 5 server that serves the catalog (live 402 prices), the picker, checkouts streamed over SSE, the ledger, the merchant feed and the live-mode gate (passcode, one live checkout at a time, daily caps).
- `@sharepay/web`: a React 19 + Vite 7 UI that walks a visitor from the catalog through the picker, the 8-step progress view, the receipt and the deliverable.

A Render Blueprint deploys both services (merchant and checkout) to Singapore.

**Architecture:**
- Server modules are small and injectable. `config` and `liveGate` are pure. `catalog`, `pickerView`, `ledgerView` and `checkouts` take their I/O as functions. `app` turns them into HTTP. `services` is the only module that builds real Binance, RPC and wallet clients.
- The web talks only to the JSON API defined in spec §6. Components receive an `Api` object, so every UI test runs on fakes in jsdom.

**Tech Stack:** Node.js >= 22.12, TypeScript ~5.9.3, Express ^5.2.1, zod ^3.25.76, viem ^2.57.3, React ^19.3.0, Vite ^7.3.7, react-markdown ^10.1.0, remark-gfm ^4.0.1, vitest ^3.2.7, supertest, jsdom + Testing Library.

**Source spec:** docs/superpowers/specs/2026-10-06-sharepay-design.md (§5 UX, §6 API, §7 config, §8 hosting)

**Conventions for every task in this plan:**
- Server code uses relative `.js` imports and workspace subpath imports (`@sharepay/buyer/checkout`, `@sharepay/core/dto`, …).
- The web uses extensionless relative imports (bundler resolution) and `import type` for DTOs.
- No test touches the internet or a real key. Server tests may bind `127.0.0.1` on an ephemeral port.
- Server gate: `npm run check -- <test file>`, which runs the whole-repo `tsc` and then that vitest file.
- Web gate: `npm run check:web -- <test file>`, which runs `tsc -p packages/web/tsconfig.json` and then that vitest file.

The live gate implements spec §6 "Live gate":
- The passcode is compared in constant time (`timingSafeEqual` on SHA-256 digests).
- An IP that fails 5 times within 10 minutes is locked out until those failures age out. More than 50 failures within an hour lock everyone out.
- Only one live checkout runs at a time.
- A slot is reserved at POST and consumed when the sale's first transaction is broadcast (`markBroadcast`, through `runCheckout`'s `onSaleBroadcast` hook). It is released if the checkout finishes before any broadcast, for example on a failed quote, simulation or price check.
- The daily count and USD caps reset at UTC midnight.

```ts
import { existsSync, readFileSync } from "node:fs";
import { dirname, isAbsolute, join, resolve } from "node:path";
import { DEFAULT_BSC_RPC_URL } from "@sharepay/core/chain/rpc";
import type { WalletMode } from "@sharepay/core/dto";
import { toUnits } from "@sharepay/core/money";
import type { Hex } from "viem";

export const CHECKOUT_VERSION = "0.1.0";

export interface CheckoutConfig {
  walletMode: WalletMode;
  buyerPrivateKey: Hex | null;
  bawBin: string;
  awPaysOwnMerchant: boolean;
  merchantUrl: string;
  livePasscode: string | null;
  liveDailyMaxCount: number;
  liveDailyMaxUsd: string;
  receiptsSeedPath: string | null;
  receiptsRuntimePath: string | null;
  webDist: string | null;
  port: number;
  rpcUrl: string;
  binance: { apiKey: string; secretKey: string } | null;
  devexEventsPath: string | null;
  version: string;
}

function repoRoot(start: string): string {
  let dir = resolve(start);
  for (;;) {
    const candidate = join(dir, "package.json");
    if (existsSync(candidate)) {
      try {
        const pkg = JSON.parse(readFileSync(candidate, "utf8")) as { name?: string };
        if (pkg.name === "sharepay") return dir;
      } catch {
        // unreadable package.json: keep walking up
      }
    }
    const parent = dirname(dir);
    if (parent === dir) return resolve(start);
    dir = parent;
  }
}

function pathEnv(env: Record<string, string | undefined>, key: string, fallback: string, root: string): string | null {
  const raw = env[key] ?? fallback;
  if (raw === "") return null;
  return isAbsolute(raw) ? raw : join(root, raw);
}

export function loadCheckoutConfig(env: Record<string, string | undefined>, cwd: string = process.cwd()): CheckoutConfig {
  const walletMode = (env.WALLET_MODE ?? "api-signer").trim();
  if (walletMode !== "api-signer" && walletMode !== "agentic-wallet") {
    throw new Error("WALLET_MODE must be api-signer or agentic-wallet");
  }
  const awPaysOwnMerchant = (env.AW_PAYS_OWN_MERCHANT ?? "true").trim() !== "false";
  const rawKey = (env.BUYER_PRIVATE_KEY ?? "").trim();
  let buyerPrivateKey: Hex | null = null;
  if (rawKey) {
    const key = rawKey.startsWith("0x") ? rawKey : `0x${rawKey}`;
    if (!/^0x[0-9a-fA-F]{64}$/.test(key)) throw new Error("BUYER_PRIVATE_KEY must be 32 bytes of hex");
    buyerPrivateKey = key as Hex;
  }
  if ((walletMode === "api-signer" || !awPaysOwnMerchant) && !buyerPrivateKey) {
    throw new Error("BUYER_PRIVATE_KEY is required for the api-signer wallet");
  }
  const merchantUrl = (env.MERCHANT_URL ?? "").trim().replace(/\/+$/, "");
  if (!/^https?:\/\/\S+$/.test(merchantUrl)) throw new Error("MERCHANT_URL must be an http(s) URL");
  const passcode = env.LIVE_PASSCODE ?? "";
  if (passcode !== "" && passcode.length < 8) throw new Error("LIVE_PASSCODE must be at least 8 characters");
  const maxCount = Number(env.LIVE_DAILY_MAX_COUNT ?? "10");
  if (!Number.isInteger(maxCount) || maxCount < 0) throw new Error("LIVE_DAILY_MAX_COUNT must be a whole number");
  const maxUsd = (env.LIVE_DAILY_MAX_USD ?? "1.50").trim();
  try {
    toUnits(maxUsd, 6);
  } catch {
    throw new Error("LIVE_DAILY_MAX_USD must be a decimal amount such as 1.50");
  }
  const port = Number(env.PORT ?? "8787");
  if (!Number.isInteger(port) || port <= 0 || port > 65535) throw new Error("PORT must be a TCP port number");
  const root = repoRoot(cwd);
  const apiKey = env.BINANCE_API_KEY ?? "";
  const secretKey = env.BINANCE_API_SECRET ?? "";
  const commit = (env.RENDER_GIT_COMMIT ?? "").trim();
  return {
    walletMode,
    buyerPrivateKey,
    bawBin: (env.BAW_BIN ?? "").trim() || "baw",
    awPaysOwnMerchant,
    merchantUrl,
    livePasscode: passcode === "" ? null : passcode,
    liveDailyMaxCount: maxCount,
    liveDailyMaxUsd: maxUsd,
    receiptsSeedPath: pathEnv(env, "RECEIPTS_SEED_PATH", "data/receipts.seed.json", root),
    receiptsRuntimePath: pathEnv(env, "RECEIPTS_RUNTIME_PATH", "data/runtime/receipts.json", root),
    webDist: pathEnv(env, "WEB_DIST", "packages/web/dist", root),
    port,
    rpcUrl: env.BSC_RPC_URL ?? DEFAULT_BSC_RPC_URL,
    binance: apiKey && secretKey ? { apiKey, secretKey } : null,
    devexEventsPath: pathEnv(env, "DEVEX_EVENTS_PATH", "", root),
    version: commit ? `${CHECKOUT_VERSION}+${commit.slice(0, 7)}` : CHECKOUT_VERSION,
  };
}
```

```ts
import { createHash, timingSafeEqual } from "node:crypto";
import type { BudgetResponse } from "@sharepay/core/dto";
import { toUnits } from "@sharepay/core/money";

export type LiveGateError = "live_disabled" | "live_passcode_invalid" | "live_locked_out" | "live_cap_reached" | "live_busy";

export type GateResult = { ok: true } | { ok: false; error: LiveGateError };

export interface LiveGate {
  budget(): BudgetResponse;
  unlock(ip: string, passcode: string): GateResult;
  reserve(ip: string, passcode: string, checkoutId: string, priceUsd: string): GateResult;
  markBroadcast(checkoutId: string): void;
  finish(checkoutId: string): void;
}

export interface LiveGateOptions {
  passcode: string | null;
  maxCount: number;
  maxUsd: string;
  nowMs: () => number;
  ipFailureLimit?: number;
  ipWindowMs?: number;
  globalFailureLimit?: number;
  globalWindowMs?: number;
}

function digest(value: string): Buffer {
  return createHash("sha256").update(value, "utf8").digest();
}

function formatMicroUsd(micros: bigint): string {
  const cents = (micros % 1_000_000n) / 10_000n;
  return `${micros / 1_000_000n}.${cents.toString().padStart(2, "0")}`;
}

export function createLiveGate(options: LiveGateOptions): LiveGate {
  const ipLimit = options.ipFailureLimit ?? 5;
  const ipWindowMs = options.ipWindowMs ?? 10 * 60 * 1000;
  const globalLimit = options.globalFailureLimit ?? 50;
  const globalWindowMs = options.globalWindowMs ?? 60 * 60 * 1000;
  const maxUsd = toUnits(options.maxUsd, 6);
  const expected = options.passcode ? digest(options.passcode) : null;
  const ipFailures = new Map<string, number[]>();
  let globalFailures: number[] = [];
  let day = "";
  let usedCount = 0;
  let usedUsd = 0n;
  let active: { id: string; usd: bigint; broadcast: boolean } | null = null;

  function rollDay(): void {
    const today = new Date(options.nowMs()).toISOString().slice(0, 10);
    if (today !== day) {
      day = today;
      usedCount = 0;
      usedUsd = 0n;
    }
  }

  function remaining(): { count: number; usd: bigint } {
    rollDay();
    const held = active && !active.broadcast ? active : null;
    const count = options.maxCount - usedCount - (held ? 1 : 0);
    const usd = maxUsd - usedUsd - (held ? held.usd : 0n);
    return { count: Math.max(count, 0), usd: usd > 0n ? usd : 0n };
  }

  function lockedOut(ip: string): boolean {
    const now = options.nowMs();
    globalFailures = globalFailures.filter((t) => now - t < globalWindowMs);
    const mine = (ipFailures.get(ip) ?? []).filter((t) => now - t < ipWindowMs);
    if (mine.length > 0) ipFailures.set(ip, mine);
    else ipFailures.delete(ip);
    return mine.length >= ipLimit || globalFailures.length >= globalLimit;
  }

  function verify(ip: string, passcode: string): GateResult {
    if (!expected) return { ok: false, error: "live_disabled" };
    if (lockedOut(ip)) return { ok: false, error: "live_locked_out" };
    if (!timingSafeEqual(digest(passcode), expected)) {
      const now = options.nowMs();
      ipFailures.set(ip, [...(ipFailures.get(ip) ?? []), now]);
      globalFailures.push(now);
      return { ok: false, error: "live_passcode_invalid" };
    }
    return { ok: true };
  }

  return {
    budget() {
      const left = remaining();
      return {
        liveAvailable: expected !== null && left.count > 0 && left.usd > 0n,
        remainingCount: left.count,
        remainingUsd: formatMicroUsd(left.usd),
      };
    },
    unlock(ip, passcode) {
      const verdict = verify(ip, passcode);
      if (!verdict.ok) return verdict;
      const left = remaining();
      return left.count > 0 && left.usd > 0n ? { ok: true } : { ok: false, error: "live_cap_reached" };
    },
    reserve(ip, passcode, checkoutId, priceUsd) {
      const verdict = verify(ip, passcode);
      if (!verdict.ok) return verdict;
      if (active) return { ok: false, error: "live_busy" };
      const price = toUnits(priceUsd, 6);
      const left = remaining();
      if (left.count < 1 || left.usd < price) return { ok: false, error: "live_cap_reached" };
      active = { id: checkoutId, usd: price, broadcast: false };
      return { ok: true };
    },
    markBroadcast(checkoutId) {
      if (!active || active.id !== checkoutId || active.broadcast) return;
      rollDay();
      active.broadcast = true;
      usedCount += 1;
      usedUsd += active.usd;
    },
    finish(checkoutId) {
      if (active?.id === checkoutId) active = null;
    },
  };
}
```

```ts
interface Slot {
  adapter: MerchantAdapter;
  own: boolean;
  entry: CatalogEntry;
  probedAt: number | null;
  inflight: Promise<void> | null;
}

function catalogItem(adapter: MerchantAdapter, status: CatalogItem["status"], statusMessage: string | null, priceUsd: string | null): CatalogItem {
  return {

    title: adapter.title,
    merchantName: adapter.merchantName,
    merchantUrl: adapter.merchantUrl,
    priceUsd,
    deliveryHint: adapter.deliveryHint,
    description: adapter.description,
    status,
    statusMessage,
  };
}

function wait(ms: number): Promise<void> {
  return new Promise((resolve) => {
    setTimeout(resolve, ms).unref();
  });
}

export function createCatalog(deps: CatalogDeps): Catalog {
  const okTtlMs = deps.okTtlMs ?? 10 * 60 * 1000;
  const retryTtlMs = deps.retryTtlMs ?? 60 * 1000;
  const ownTimeoutMs = deps.ownProbeTimeoutMs ?? 90_000;
  const thirdPartyTimeoutMs = deps.thirdPartyProbeTimeoutMs ?? 45_000;
  const ownAttempts = deps.ownAttempts ?? 3;
  const ownRetryDelayMs = deps.ownRetryDelayMs ?? 5_000;
  const firstWaitMs = deps.firstWaitMs ?? 3_000;
  const preferredAssets = [getStable("U").address, getStable("USD1").address];
  const slots = new Map<ItemId, Slot>();
  for (const adapter of deps.adapters) {
    const own = adapter.id === deps.ownItemId;
    slots.set(adapter.id, {
      adapter,
      own,
      entry: { item: catalogItem(adapter, "waking", own ? WAKING_MESSAGE : "Checking the price…", null), priceAtomic: null, payAsset: null },
      probedAt: null,
      inflight: null,
    });
  }

  async function probe(slot: Slot): Promise<void> {
    const attempts = slot.own ? ownAttempts : 1;
    const timeoutMs = slot.own ? ownTimeoutMs : thirdPartyTimeoutMs;
    for (let attempt = 1; attempt <= attempts; attempt += 1) {
      try {
        const { selected } = await fetchChallenge(deps.fetchImpl, { ...slot.adapter.request(), timeoutMs }, { maxAmount: MAX_PAYMENT_ATOMIC, preferredAssets });
        const priceAtomic = BigInt(selected.accept.amount);
        slot.entry = { item: catalogItem(slot.adapter, "ok", null, formatPriceUsd(priceAtomic)), priceAtomic, payAsset: selected.stable.address };
        return;
      } catch {
        if (attempt < attempts) await deps.sleep(ownRetryDelayMs);
      }
    }
    slot.entry = slot.own
      ? { item: catalogItem(slot.adapter, "waking", WAKING_MESSAGE, null), priceAtomic: null, payAsset: null }
      : { item: catalogItem(slot.adapter, "unavailable", UNAVAILABLE_MESSAGE, null), priceAtomic: null, payAsset: null };
  }

  function stale(slot: Slot): boolean {
    if (slot.probedAt === null) return true;
    const age = deps.nowMs() - slot.probedAt;
    if (slot.entry.item.status === "ok") return age >= okTtlMs;
    return slot.own || age >= retryTtlMs;
  }

  function ensureProbe(slot: Slot): Promise<void> {
    if (slot.inflight) return slot.inflight;
    if (!stale(slot)) return Promise.resolve();
    if (slot.own) {
      deps.fetchImpl(`${deps.merchantUrl}/ping`, { signal: AbortSignal.timeout(ownTimeoutMs) }).catch(() => undefined);
    }
    const run = probe(slot).finally(() => {
      slot.probedAt = deps.nowMs();
      slot.inflight = null;
    });
    slot.inflight = run;
    return run;
  }

  return {
    async list() {
      const blocking: Promise<void>[] = [];
      for (const slot of slots.values()) {
        const run = ensureProbe(slot);
        if (slot.entry.item.status !== "ok") blocking.push(run);
      }
      if (blocking.length > 0) await Promise.race([Promise.all(blocking), wait(firstWaitMs)]);
      return [...slots.values()].map((slot) => slot.entry.item);
    },
    async entry(itemId) {
      const slot = slots.get(itemId);
      if (!slot) return null;
      const run = ensureProbe(slot);
      if (slot.entry.item.status !== "ok") await run;
      return slot.entry;
    },
    adapter: (itemId) => slots.get(itemId)?.adapter ?? null,
    async settle() {
      await Promise.all([...slots.values()].map((slot) => slot.inflight ?? Promise.resolve()));
    },
  };
}
```

```ts
export function createPickerView(deps: PickerViewDeps): PickerView {
  const ttlMs = deps.ttlMs ?? 30_000;
  const cache = new Map<ItemId, { at: number; value: Promise<PickerResponse> }>();

  async function compute(itemId: ItemId): Promise<PickerResponse> {
    const entry = await deps.catalog.entry(itemId);
    if (!entry || entry.item.status !== "ok" || entry.priceAtomic === null || entry.payAsset === null) {
      throw new ItemUnavailableError(itemId);
    }
    const owner = await deps.walletFor(itemId).address();
    const holdings = await deps.holdings(owner, await deps.snapshot());
    const options = await computePicker({ probe: deps.probe }, { holdings, priceAtomic: entry.priceAtomic, payAsset: entry.payAsset, wallet: owner });
    return { itemId, priceUsd: entry.item.priceUsd ?? "0.00", options };
  }

  return {
    get(itemId) {
      const cached = cache.get(itemId);
      if (cached && deps.nowMs() - cached.at < ttlMs) return cached.value;
      const value = compute(itemId);
      cache.set(itemId, { at: deps.nowMs(), value });
      value.catch(() => {
        if (cache.get(itemId)?.value === value) cache.delete(itemId);
      });
      return value;
    },
```

```ts
import type { CheckoutRequest } from "@sharepay/buyer/checkout";
import { initialSteps, type CheckoutState } from "@sharepay/core/dto";

export type RunCheckout = (request: CheckoutRequest, onSaleBroadcast: () => void, emit: (state: CheckoutState) => void) => Promise<CheckoutState>;

export interface CheckoutHooks {
  onSaleBroadcast?: () => void;
  onFinish?: (state: CheckoutState) => void;
}

export interface CheckoutRegistry {
  start(request: CheckoutRequest, hooks?: CheckoutHooks): void;
  get(id: string): CheckoutState | null;
  subscribe(id: string, listener: (state: CheckoutState) => void): () => void;
  running(): number;
}

export function isTerminal(state: CheckoutState): boolean {
  return state.status === "succeeded" || state.status === "failed";
}

interface Entry {
  state: CheckoutState | null;
  listeners: Set<(state: CheckoutState) => void>;
}

export function createCheckoutRegistry(deps: { run: RunCheckout; nowMs: () => number; maxKept?: number }): CheckoutRegistry {
  const maxKept = deps.maxKept ?? 200;
  const entries = new Map<string, Entry>();

  function evict(): void {
    for (const [id, entry] of entries) {
      if (entries.size <= maxKept) return;
      if (entry.state && isTerminal(entry.state)) entries.delete(id);
    }
  }

  function crashed(request: CheckoutRequest, last: CheckoutState | null, error: unknown): CheckoutState {
    const now = new Date(deps.nowMs()).toISOString();
    const base: CheckoutState = last ?? {
      id: request.id,
      itemId: request.adapter.id,
      mode: request.mode,
      tokenAddress: request.tokenAddress,
      status: "failed",
      steps: initialSteps(now),
      receipt: null,
      deliverable: null,
      error: null,
      etaSec: null,
      jobStartedAt: null,
      lastPolledAt: null,
      startedAt: now,
      finishedAt: null,
    };
    const message = error instanceof Error ? error.message : String(error);
    return { ...base, status: "failed", etaSec: null, error: { code: "unexpected_error", message }, finishedAt: now };
  }

  return {
    start(request, hooks = {}) {
      if (entries.has(request.id)) throw new Error(`checkout ${request.id} already exists`);
      const entry: Entry = { state: null, listeners: new Set() };
      entries.set(request.id, entry);
      let finished = false;
      const publish = (state: CheckoutState) => {
        entry.state = state;
        for (const listener of [...entry.listeners]) listener(state);
      };
      const finish = (state: CheckoutState) => {
        if (finished) return;
        finished = true;
        publish(state);
        hooks.onFinish?.(state);
        evict();
      };
      deps
        .run(request, () => hooks.onSaleBroadcast?.(), (state) => {
          if (!finished) publish(state);
        })
        .then(finish, (error: unknown) => finish(crashed(request, entry.state, error)));
    },
    get: (id) => entries.get(id)?.state ?? null,
    subscribe(id, listener) {
      const entry = entries.get(id);
      if (!entry) return () => undefined;
      entry.listeners.add(listener);
      return () => {
        entry.listeners.delete(listener);
      };
    },
    running() {
      let count = 0;
      for (const entry of entries.values()) if (!entry.state || !isTerminal(entry.state)) count += 1;
      return count;
    },
  };
}
```

```ts
function catalogItem(id: ItemId, status: CatalogItem["status"]): CatalogItem {
  return { id, title: id, merchantName: "m", merchantUrl: "https://m", priceUsd: status === "ok" ? "0.05" : null, deliveryHint: "Instant", description: "d", status, statusMessage: null };
}

function deferred(): { promise: Promise<void>; resolve: () => void } {
  let resolve: () => void = () => undefined;
  const promise = new Promise<void>((r) => {
    resolve = r;
  });
  return { promise, resolve };
}
```

```ts
import { existsSync } from "node:fs";
import { join } from "node:path";
import type { CheckoutRequest } from "@sharepay/buyer/checkout";
import type { CheckoutState, HealthResponse, ItemId, LedgerResponse, MerchantActivity, PickerResponse } from "@sharepay/core/dto";
import express, { type Express, type NextFunction, type Request, type Response } from "express";
import type { Address } from "viem";
import { z } from "zod";
import type { Catalog } from "./catalog.js";
import { isTerminal, type CheckoutRegistry } from "./checkouts.js";
import type { LiveGate, LiveGateError } from "./liveGate.js";
import { ItemUnavailableError } from "./pickerView.js";

export const ITEM_IDS = ["sharepay-nvda-report", "bnb-stock-analyst-nvda", "cmc-nvda-dossier"] as const;

export interface AppDeps {
  health: () => HealthResponse;
  deepHealth: () => Promise<Record<string, unknown>>;
  catalog: Pick<Catalog, "list" | "entry" | "adapter">;
  picker: { get(itemId: ItemId): Promise<PickerResponse> };
  ledger: { get(): Promise<LedgerResponse> };
  merchantActivity: () => Promise<MerchantActivity[]>;
  gate: LiveGate;
  checkouts: CheckoutRegistry;
  newId: () => string;
  webDist: string | null;
  maxRunning?: number;
  ssePingMs?: number;
}

const ItemIdSchema = z.enum(ITEM_IDS);
const CheckoutBody = z.object({
  itemId: ItemIdSchema,
  tokenAddress: z.string().regex(/^0x[0-9a-fA-F]{40}$/),
  mode: z.enum(["simulate", "live"]),
  passcode: z.string().max(256).optional(),
});
const UnlockBody = z.object({ passcode: z.string().max(256) });

function gateStatus(error: LiveGateError): number {
  return error === "live_busy" ? 409 : 403;
}

function clientIp(req: Request): string {
  return req.ip ?? "unknown";
}

export function createApp(deps: AppDeps): Express {
  const app = express();
  const maxRunning = deps.maxRunning ?? 4;
  const pingMs = deps.ssePingMs ?? 20_000;
  app.set("trust proxy", 1);
  app.disable("x-powered-by");
  app.use(express.json({ limit: "10kb" }));
  app.use((_req, res, next) => {
    res.set("X-Content-Type-Options", "nosniff");
    next();
  });

  app.get("/api/health", async (req, res) => {
    const base = deps.health();
    if (req.query.deep === "1") {
      res.json({ ...base, ...(await deps.deepHealth()) });
      return;
    }
    res.json(base);
  });

  app.get("/api/catalog", async (_req, res) => {
    res.json(await deps.catalog.list());
  });

  app.get("/api/picker", async (req, res) => {
    const parsed = ItemIdSchema.safeParse(req.query.itemId);
    if (!parsed.success) {
      res.status(400).json({ error: "bad_request" });
      return;
    }
    try {
      res.json(await deps.picker.get(parsed.data));
    } catch (error) {
      if (error instanceof ItemUnavailableError) {
        res.status(503).json({ error: "item_unavailable" });
        return;
      }
      res.status(502).json({ error: "picker_failed", detail: error instanceof Error ? error.message : String(error) });
    }
  });

  app.post("/api/live/unlock", (req, res) => {
    const parsed = UnlockBody.safeParse(req.body);
    if (!parsed.success) {
      res.status(400).json({ error: "bad_request" });
      return;
    }
    const verdict = deps.gate.unlock(clientIp(req), parsed.data.passcode);
    if (!verdict.ok) {
      res.status(gateStatus(verdict.error)).json({ error: verdict.error });
      return;
    }
    res.json(deps.gate.budget());
  });

  app.get("/api/budget", (_req, res) => {
    res.json(deps.gate.budget());
  });

  app.post("/api/checkout", async (req, res) => {
    const parsed = CheckoutBody.safeParse(req.body);
    if (!parsed.success) {
      res.status(400).json({ error: "bad_request" });
      return;
    }
    const body = parsed.data;
    const entry = await deps.catalog.entry(body.itemId);
    const adapter = deps.catalog.adapter(body.itemId);
    if (!entry || entry.item.status !== "ok" || !adapter) {
      res.status(503).json({ error: "item_unavailable" });
      return;
    }
    if (deps.checkouts.running() >= maxRunning) {
      res.status(429).json({ error: "too_many_checkouts" });
      return;
    }
    const id = deps.newId();
    if (body.mode === "live") {
      const verdict = deps.gate.reserve(clientIp(req), body.passcode ?? "", id, entry.item.priceUsd ?? "0");
      if (!verdict.ok) {
        res.status(gateStatus(verdict.error)).json({ error: verdict.error });
        return;
      }
    }
    const request: CheckoutRequest = { id, adapter, tokenAddress: body.tokenAddress as Address, mode: body.mode };
    deps.checkouts.start(
      request,
      body.mode === "live" ? { onSaleBroadcast: () => deps.gate.markBroadcast(id), onFinish: () => deps.gate.finish(id) } : {},
    );
    res.status(202).json({ checkoutId: id });
  });

  app.get("/api/checkout/:id", (req, res) => {
    const state = deps.checkouts.get(req.params.id);
    if (!state) {
      res.status(404).json({ error: "unknown_checkout" });
      return;
    }
    res.json(state);
  });

  app.get("/api/checkout/:id/events", (req, res) => {
    const id = req.params.id;
    const current = deps.checkouts.get(id);
    if (!current) {
      res.status(404).json({ error: "unknown_checkout" });
      return;
    }
    res.status(200).set({
      "Content-Type": "text/event-stream; charset=utf-8",
      "Cache-Control": "no-cache, no-transform",
      Connection: "keep-alive",
      "X-Accel-Buffering": "no",
    });
    res.flushHeaders();
    let done = false;
    let unsubscribe: () => void = () => undefined;
    const ping = setInterval(() => {
      res.write(": ping\n\n");
    }, pingMs);
    const finish = () => {
      if (done) return;
      done = true;
      clearInterval(ping);
      unsubscribe();
      res.end();
    };
    const send = (state: CheckoutState) => {
      if (done) return;
      res.write(`event: state\ndata: ${JSON.stringify(state)}\n\n`);
      if (isTerminal(state)) finish();
    };
    res.on("close", finish);
    send(current);
    if (!done) unsubscribe = deps.checkouts.subscribe(id, send);
  });

  app.get("/api/ledger", async (_req, res) => {
    res.json(await deps.ledger.get());
  });

  app.get("/api/merchant/activity", async (_req, res) => {
    try {
      res.json({ items: await deps.merchantActivity() });
    } catch {
      res.json({ items: [] });
    }
  });

  app.use("/api", (_req, res) => {
    res.status(404).json({ error: "not_found" });
  });

  const dist = deps.webDist;
  if (dist && existsSync(join(dist, "index.html"))) {
    app.use(express.static(dist));
    app.use((req, res, next) => {
      if (req.method !== "GET") {
        next();
        return;
      }
      res.sendFile(join(dist, "index.html"));
    });
  } else {
    app.get("/", (_req, res) => {
      res.type("text/plain").send("SharePay checkout API. The web app is not built yet (npm run build:web).");
    });
  }

  app.use((error: unknown, _req: Request, res: Response, _next: NextFunction) => {
    const type = (error as { type?: string } | null)?.type;
    if (type === "entity.parse.failed" || type === "entity.too.large") {
      res.status(400).json({ error: "bad_request" });
      return;
    }
    res.status(500).json({ error: "internal_error" });
  });

  return app;
}
```

```ts
export function memoize<T>(fn: () => Promise<T>, ttlMs: number, nowMs: () => number): () => Promise<T> {
  let cached: { at: number; value: Promise<T> } | null = null;
  return () => {
    if (cached && nowMs() - cached.at < ttlMs) return cached.value;
    const entry = { at: nowMs(), value: fn() };
    cached = entry;
    entry.value.catch(() => {
      if (cached === entry) cached = null;
    });
    return entry.value;
  };
}

export function receiptVerifier(chain: Pick<ChainReader, "waitForReceipt">): (receipt: Receipt) => Promise<boolean> {
  return async (receipt) => {
    const hash = receipt.settlementTxHash ?? receipt.swapTxHashes[0];
    if (!hash || !/^0x[0-9a-fA-F]{64}$/.test(hash)) return false;
    const mined = await chain.waitForReceipt(hash as Hex, 10_000);
    return mined.status === "success";
  };
}
```

```ts
}

function errorCode(body: unknown, status: number): string {
  if (body && typeof body === "object" && "error" in body) {

async function readBody(response: Response): Promise<unknown> {
  const text = await response.text();
  if (!text) return null;
  try {
    return JSON.parse(text) as unknown;
  } catch {

export function createApi(fetchImpl?: typeof fetch, baseUrl?: string): Api {
  const fetchFn = fetchImpl ?? globalThis.fetch.bind(globalThis);
  const base = baseUrl ?? "";

  async function call(path: string, init?: RequestInit): Promise<{ status: number; body: unknown }> {
    const response = init ? await fetchFn(`${base}${path}`, init) : await fetchFn(`${base}${path}`);
    const body = await readBody(response);
    return { status: response.status, body };
  }

  function ok<T>(status: number, body: unknown): T {
    if (status < 200 || status >= 300) {
      throw new ApiError(status, errorCode(body, status));
    }
    return body as T;
  }

  function post(body: unknown): RequestInit {
    return {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(body),
    };
  }

  return {
    async health() {
      const { status, body } = await call("/api/health");
      return ok(status, body);
    },
    async catalog() {
      const { status, body } = await call("/api/catalog");
      return ok(status, body);
    },
    async picker(itemId) {
      const { status, body } = await call(`/api/picker?itemId=${encodeURIComponent(itemId)}`);
      return ok(status, body);
    },
    async unlock(passcode) {
      const { status, body } = await call("/api/live/unlock", post({ passcode }));
      return ok(status, body);
    },
    async budget() {
      const { status, body } = await call("/api/budget");
      return ok(status, body);
    },
    async startCheckout(body) {
      const { status, body: parsed } = await call("/api/checkout", post(body));
      return ok(status, parsed);
    },
    async checkout(id) {
      const { status, body } = await call(`/api/checkout/${encodeURIComponent(id)}`);
      if (status === 404) return null;
      return ok(status, body);

      const { status, body } = await call("/api/ledger");
      return ok(status, body);

    async merchantActivity() {
      const { status, body } = await call("/api/merchant/activity");
```

```ts
import type { CheckoutState } from "@sharepay/core/dto";
import type { Api } from "./api";
import { isTerminal } from "./format";

export interface FollowHandlers {
  onState(state: CheckoutState): void;
  onGone(): void;
}

export interface FollowDeps {
  api: Pick<Api, "checkout">;
  EventSourceImpl?: typeof EventSource;
  baseUrl?: string;
  setTimeoutImpl?: (fn: () => void, ms: number) => unknown;
  clearTimeoutImpl?: (handle: unknown) => void;
}

export const BACKOFF_MS = [1000, 2000, 4000, 8000, 15000];

export function followCheckout(
  id: string,
  handlers: FollowHandlers,
  deps: FollowDeps,
): { close(): void } {
  const baseUrl = deps.baseUrl ?? "";
  const EventSourceImpl = deps.EventSourceImpl ?? globalThis.EventSource;
  const setTimeoutImpl = deps.setTimeoutImpl ?? ((fn, ms) => setTimeout(fn, ms));
  const clearTimeoutImpl =
    deps.clearTimeoutImpl ?? ((handle) => clearTimeout(handle as ReturnType<typeof setTimeout>));

  let stopped = false;
  let attempt = 0;
  let source: EventSource | null = null;
  let timer: unknown = null;
  let generation = 0;

  const clearTimer = () => {
    if (timer == null) return;
    clearTimeoutImpl(timer);
    timer = null;
  };

  const stop = () => {
    stopped = true;
    generation += 1;
    if (source) {
      const current = source;
      source = null;
      current.close();
    }
    clearTimer();
  };

  const schedule = () => {
    if (stopped) return;
    const delay = BACKOFF_MS[Math.min(attempt, BACKOFF_MS.length - 1)] ?? BACKOFF_MS[BACKOFF_MS.length - 1] ?? 15000;
    attempt += 1;
    clearTimer();
    timer = setTimeoutImpl(() => {
      timer = null;
      if (stopped) return;
      connect();
    }, delay);
  };

  const fallback = async (gen: number) => {
    let state: CheckoutState | null;
    try {
      state = await deps.api.checkout(id);
    } catch {
      if (stopped || gen !== generation) return;
      schedule();
      return;
    }
    if (stopped || gen !== generation) return;
    if (state == null) {
      handlers.onGone();
      stop();
      return;
    }
    handlers.onState(state);
    if (stopped || gen !== generation) return;
    if (isTerminal(state.status)) {
      stop();
      return;
    }
    schedule();
  };

  const connect = () => {
    if (stopped) return;
    if (source) {
      const previous = source;
      source = null;
      previous.close();
    }
    const es = new EventSourceImpl(`${baseUrl}/api/checkout/${encodeURIComponent(id)}/events`);
    source = es;

    es.addEventListener("state", (event) => {
      if (stopped || source !== es) return;
      let state: CheckoutState;
      try {
        state = JSON.parse(String((event as MessageEvent).data)) as CheckoutState;
      } catch {
        return;
      }
      attempt = 0;
      handlers.onState(state);
      if (isTerminal(state.status)) stop();
    });

    es.addEventListener("error", () => {
      if (stopped || source !== es) return;
      source = null;
      es.close();
      const gen = generation;
      void fallback(gen);
    });
  };

  connect();

  return {
    close() {
      stop();
    },
  };
}
```

```ts
  const date = utcParts(iso);
  const hh = String(date.getUTCHours()).padStart(2, "0");
  const mm = String(date.getUTCMinutes()).padStart(2, "0");
  const ss = String(date.getUTCSeconds()).padStart(2, "0");
  return `${hh}:${mm}:${ss} UTC`;
}

export function fmtDateTime(iso: string): string {
  const date = utcParts(iso);
  const month = MONTHS[date.getUTCMonth()] ?? "Jan";
  const day = date.getUTCDate();
  const hh = String(date.getUTCHours()).padStart(2, "0");
  const mm = String(date.getUTCMinutes()).padStart(2, "0");
  return `${month} ${day}, ${hh}:${mm} UTC`;
}

export function shortAddress(address: string): string {
  if (address.length <= 10) return address;
  return `${address.slice(0, 6)}…${address.slice(-4)}`;
}

export function isTerminal(status: CheckoutStatus): boolean {
  return status === "succeeded" || status === "failed";
}

export function latestReceiptFor(receipts: Receipt[], itemId: ItemId): Receipt | null {
  let best: Receipt | null = null;
  let bestTime = Number.NEGATIVE_INFINITY;
  for (const receipt of receipts) {
    if (receipt.itemId !== itemId) continue;
    const time = Date.parse(receipt.createdAt);
    if (best == null || time > bestTime) {
      best = receipt;
      bestTime = time;
    }
  }
  return best;
}
```

```tsx
import { useEffect, useRef, useState, type FormEvent } from "react";
import { ApiError, type Api } from "../api";
import { errorCopy } from "../copy";

export function UnlockDialog({
  api,
  onUnlocked,
  onCancel,
}: {
  api: Pick<Api, "unlock">;
  onUnlocked: (passcode: string, budget: BudgetResponse) => void;
  onCancel: () => void;
}) {
  const [passcode, setPasscode] = useState("");
  const [pending, setPending] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const active = useRef(true);

  useEffect(() => {
    active.current = true;
    return () => {
      active.current = false;
    };
  }, []);

  async function onSubmit(event: FormEvent) {
    event.preventDefault();
    if (pending) return;
    setPending(true);
    setError(null);
    try {
      const budget = await api.unlock(passcode);
      if (!active.current) return;
      onUnlocked(passcode, budget);
    } catch (err) {
      if (!active.current) return;
      const code = err instanceof ApiError ? err.code : "unknown";
      setError(errorCopy(code));
    } finally {
      if (active.current) setPending(false);
    }
  }

  return (
    <div className="dialog-backdrop">
      <div role="dialog" aria-label="Unlock live mode" className="card dialog">
        <h2>Unlock live mode</h2>
```

```tsx
import type { HealthResponse } from "@sharepay/core/dto";
import { useEffect } from "react";
import type { Api } from "../api";

export function BootScreen({
  api,
  onReady,
  pollMs = 2000,
}: {
  api: Pick<Api, "health">;
  onReady: (health: HealthResponse) => void;
  pollMs?: number;
}) {
  useEffect(() => {
    let stopped = false;
    let timer: ReturnType<typeof setTimeout> | undefined;

    const tick = async () => {
      try {
        const health = await api.health();
        if (stopped) return;
        if (health.ok) {
          onReady(health);
          return;
        }
      } catch {
        if (stopped) return;
      }
      if (stopped) return;
      timer = setTimeout(() => {
        void tick();
      }, pollMs);
    };

    void tick();

    return () => {
      stopped = true;
      if (timer !== undefined) clearTimeout(timer);
    };
  }, [api, onReady, pollMs]);

  return (
    <section className="boot">
      <p>Starting SharePay…</p>
      <p>The free host wakes up in about a minute.</p>
    </section>
  );
}
```

```tsx
import type { CheckoutMode, PickerOption, PickerResponse } from "@sharepay/core/dto";
import { useRef, useState } from "react";
import { fmtUsd } from "../format";

function preferredAddress(options: PickerOption[]): string | null {
  const ranked = options.find((option) => option.payable && option.rank === 1);
  if (ranked) return ranked.tokenAddress;
  return options.find((option) => option.payable)?.tokenAddress ?? null;
}

function optionLabel(option: PickerOption): string {
  return `${option.name} · ${option.sharesHeld} shares (~${fmtUsd(option.valueUsd)})`;
}

export function Picker({
  response,
  mode,
  onPay,
  busy,
}: {
  response: PickerResponse;
  mode: CheckoutMode;
  onPay: (option: PickerOption) => void;
  busy: boolean;
}) {
  const preferred = preferredAddress(response.options);
  const [selected, setSelected] = useState<string | null>(preferred);
  const seen = useRef(response);
  if (seen.current !== response) {
    seen.current = response;
    setSelected(preferred);
  }

  const selectedOption =
    response.options.find((option) => option.tokenAddress === selected && option.payable && option.plan) ?? null;

  return (
    <section className="picker">
      <div role="radiogroup" aria-label="Pay with">
        {response.options.map((option) => (
          <div key={option.tokenAddress} className={option.payable ? "option" : "option greyed"}>
            <label>
              <input
                type="radio"
                name="pay-with"
                value={option.tokenAddress}
                checked={selected === option.tokenAddress}
                disabled={!option.payable}
                onChange={() => setSelected(option.tokenAddress)}
              />
              {optionLabel(option)}
            </label>
            {option.badges.map((badge, index) => (
              <span key={`${badge.text}-${index}`} className={`badge ${badge.tone}`}>
                {badge.text}
              </span>
            ))}
            {!option.payable && option.reason && <span className="reason">{option.reason.message}</span>}
          </div>
        ))}
      </div>
      {selectedOption?.plan ? (
        <button type="button" disabled={busy} onClick={() => onPay(selectedOption)}>
          {mode === "live"
            ? `Pay ${fmtUsd(response.priceUsd)} with ≈${selectedOption.plan.sellShares} ${selectedOption.name} shares`
            : `Simulate paying ${fmtUsd(response.priceUsd)} with ≈${selectedOption.plan.sellShares} ${selectedOption.name} shares`}
        </button>
      ) : (
        <p>None of your stocks can pay for this right now.</p>
      )}
    </section>
  );
}
```

```tsx
import type { CheckoutState, StepStatus } from "@sharepay/core/dto";
import { useState } from "react";
import { fmtTime } from "../format";

const GLYPH: Record<StepStatus, string> = {
  pending: "○",
  running: "●",
  done: "✓",
  skipped: "–",
  failed: "✕",
};

function etaLine(etaSec: number | null): string {
  if (etaSec == null) return "Report generating";
  return `Report generating — about ${Math.ceil(etaSec / 60)} min left`;
}

function pollLine(jobStartedAt: string | null, lastPolledAt: string | null): string | null {
  if (jobStartedAt && lastPolledAt) {
    return `Started ${fmtTime(jobStartedAt)} · last checked ${fmtTime(lastPolledAt)}`;
  }
  if (jobStartedAt) return `Started ${fmtTime(jobStartedAt)}`;
  if (lastPolledAt) return `last checked ${fmtTime(lastPolledAt)}`;
  return null;
}

export function Progress({ state }: { state: CheckoutState }) {
  const [expanded, setExpanded] = useState<Record<string, boolean>>({});
  const poll = state.status === "awaiting_delivery" ? pollLine(state.jobStartedAt, state.lastPolledAt) : null;

  return (
    <section className="progress">
      <ol className="steps">
        {state.steps.map((step) => {
          const open = expanded[step.id] === true;
          return (
            <li key={step.id} className={`step step-${step.status}`}>
              <span className={`step-icon step-${step.status}`} role="img" aria-label={step.status}>
                {GLYPH[step.status]}
              </span>
              <p className="step-title">{step.title}</p>
              <p className="step-summary">{step.summary}</p>
              <button
                type="button"
                className="step-toggle"
                aria-expanded={open}
                onClick={() => setExpanded((current) => ({ ...current, [step.id]: !current[step.id] }))}
              >
                {open ? "Hide details" : "Details"}
              </button>
              {open && (
                <div className="step-detail">
                  <p>{step.technicalTitle}</p>
                  <dl>
                    {Object.entries(step.detail).map(([key, value]) => (
                      <div key={key}>
                        <dt>{key}</dt>
                        <dd>{value}</dd>
                      </div>
                    ))}
                  </dl>
                  {step.links.map((link) => (
                    <a key={`${link.href}-${link.label}`} href={link.href} target="_blank" rel="noopener noreferrer">
                      {link.label}
                    </a>
                  ))}
                </div>
              )}
            </li>
          );
        })}
      </ol>
      {state.status === "awaiting_delivery" && <p>{etaLine(state.etaSec)}</p>}
      {poll && <p>{poll}</p>}
      {state.status === "failed" && state.error && (
        <div role="alert" className="alert">
          <p>{state.error.message}</p>
          <code>{state.error.code}</code>
        </div>
      )}
    </section>
  );
}
```

```tsx
import type { MerchantActivity } from "@sharepay/core/dto";
import { useEffect, useState } from "react";
import type { Api } from "../api";
import { fmtDateTime } from "../format";

export function MerchantFeed({
  api,
  pollMs = 10000,
}: {
  api: Pick<Api, "merchantActivity">;
  pollMs?: number;
}) {
  const [items, setItems] = useState<MerchantActivity[] | null>(null);

  useEffect(() => {
    let stopped = false;
    let timer: ReturnType<typeof setTimeout> | undefined;

    const load = async () => {
      try {
        const next = await api.merchantActivity();
        if (!stopped && Array.isArray(next)) setItems(next);
      } catch {
        if (!stopped) setItems((current) => current ?? []);
      }
      if (stopped) return;
      timer = setTimeout(() => {
        void load();
      }, pollMs);
    };

    void load();

    return () => {
      stopped = true;
      if (timer !== undefined) clearTimeout(timer);
    };
  }, [api, pollMs]);

  const sorted = [...(items ?? [])].sort((a, b) => Date.parse(b.at) - Date.parse(a.at));

  return (
    <section className="merchant-feed">
      <h2>Merchant agent</h2>
      {items == null ? null : sorted.length === 0 ? (
        <p>No merchant activity yet.</p>
      ) : (
        <ul>
          {sorted.map((item) => (
            <li key={item.id}>
              <p>{item.summary}</p>
              <p>{fmtDateTime(item.at)}</p>
```

```tsx
import type {
  BudgetResponse,
  CatalogItem,
  CheckoutMode,
  CheckoutState,
  HealthResponse,
  LedgerResponse,
  PickerOption,
  PickerResponse,
  Receipt,
} from "@sharepay/core/dto";
import { useCallback, useEffect, useRef, useState } from "react";
import { ApiError, type Api } from "./api";
import { BootScreen } from "./components/BootScreen";
import { Catalog } from "./components/Catalog";
import { DeliverableView } from "./components/DeliverableView";
import { Header } from "./components/Header";
import { LedgerView } from "./components/LedgerView";
import { MerchantFeed } from "./components/MerchantFeed";
import { Picker } from "./components/Picker";
import { Progress } from "./components/Progress";
import { ReceiptCard } from "./components/ReceiptCard";
import { RecordedRun } from "./components/RecordedRun";
import { UnlockDialog } from "./components/UnlockDialog";
import { errorCopy } from "./copy";
import { followCheckout, type FollowDeps, type FollowHandlers } from "./follow";
import { isTerminal, latestReceiptFor } from "./format";

const STORAGE_KEY = "sharepay.checkoutId";

type AppStorage = Pick<Storage, "getItem" | "setItem" | "removeItem">;
type View = "catalog" | "picker" | "progress" | "missing" | "resuming";
type Tab = "checkout" | "ledger";

export function App({
  api,
  followImpl = followCheckout,
  storage = window.localStorage,
}: {
  api: Api;
  followImpl?: (id: string, handlers: FollowHandlers, deps: FollowDeps) => { close(): void };
  storage?: AppStorage;
}) {
  const [health, setHealth] = useState<HealthResponse | null>(null);
  const [tab, setTab] = useState<Tab>("checkout");
  const [mode, setMode] = useState<CheckoutMode>("simulate");
  const [passcode, setPasscode] = useState<string | null>(null);
  const [budget, setBudget] = useState<BudgetResponse | null>(null);
  const [unlockOpen, setUnlockOpen] = useState(false);
  const [items, setItems] = useState<CatalogItem[]>([]);
  const [view, setView] = useState<View>("catalog");
  const [resumeId, setResumeId] = useState<string | null>(null);
  const [selected, setSelected] = useState<CatalogItem | null>(null);
  const [picker, setPicker] = useState<PickerResponse | null>(null);
  const [pickerLoading, setPickerLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [busy, setBusy] = useState(false);
  const [checkoutState, setCheckoutState] = useState<CheckoutState | null>(null);
  const [endedMessage, setEndedMessage] = useState(false);
  const [recorded, setRecorded] = useState<Receipt | null | undefined>(undefined);
  const [ledger, setLedger] = useState<LedgerResponse | null>(null);
  const [ledgerError, setLedgerError] = useState<string | null>(null);

  const followRef = useRef<{ close(): void } | null>(null);
  const mountedRef = useRef(true);
  const sessionRef = useRef(0);
  const apiRef = useRef(api);
  const followImplRef = useRef(followImpl);
  apiRef.current = api;
  followImplRef.current = followImpl;

  useEffect(() => {
    mountedRef.current = true;
    return () => {
      mountedRef.current = false;
      followRef.current?.close();
      followRef.current = null;
    };
  }, []);

  const loadRecorded = useCallback(async (itemId: CheckoutState["itemId"]) => {
    try {
      const data = await apiRef.current.ledger();
      if (!mountedRef.current) return;
      setRecorded(latestReceiptFor(data.receipts, itemId));
    } catch {
      if (!mountedRef.current) return;
      setRecorded(null);
    }
  }, []);

  const beginFollow = useCallback((id: string) => {
    followRef.current?.close();
    const session = sessionRef.current;
    const handle = followImplRef.current(
      id,
      {
        onState(state) {
          if (!mountedRef.current || session !== sessionRef.current) return;
          setCheckoutState(state);
          if (state.status === "succeeded" && state.mode === "simulate") {
            void loadRecorded(state.itemId);
          }
        },
        onGone() {
          if (!mountedRef.current || session !== sessionRef.current) return;
          setEndedMessage(true);
        },
      },
      { api: apiRef.current },
    );
    followRef.current = handle;
  }, [loadRecorded]);

  const onReady = useCallback(
    (next: HealthResponse) => {
      const existing = storage.getItem(STORAGE_KEY);
      if (existing) {
        setResumeId(existing);
        setView("resuming");
      }
      setHealth(next);
    },
    [storage],
  );

  useEffect(() => {
    if (!health) return;
    let stopped = false;
    let timer: ReturnType<typeof setTimeout> | undefined;

    const load = async () => {
      try {
        const next = await api.catalog();
        if (stopped) return;
        setItems(next);
        if (next.some((item) => item.status === "waking")) {
          timer = setTimeout(() => {
            void load();
          }, 5000);
        }
      } catch {
        if (stopped) return;
        timer = setTimeout(() => {
          void load();
        }, 5000);
      }
    };

    void load();
    return () => {
      stopped = true;
      if (timer !== undefined) clearTimeout(timer);
    };
  }, [api, health]);

  useEffect(() => {
    if (!resumeId) return;
    let stopped = false;

    void (async () => {
      try {
        const state = await api.checkout(resumeId);
        if (stopped) return;
        if (state == null) {
          setCheckoutState(null);
          setEndedMessage(true);
          setView("missing");
          return;
        }
        setCheckoutState(state);
        setView("progress");
        if (!isTerminal(state.status)) {
          beginFollow(resumeId);
        } else if (state.status === "succeeded" && state.mode === "simulate") {
          void loadRecorded(state.itemId);
        }
      } catch (err) {
        if (stopped) return;
        setError(err instanceof ApiError ? errorCopy(err.code) : errorCopy("unknown"));
        setEndedMessage(true);
        setView("missing");
      }
    })();

    return () => {
      stopped = true;
      followRef.current?.close();
      followRef.current = null;
    };
  }, [resumeId, api, beginFollow, loadRecorded]);

  useEffect(() => {
    if (!health || tab !== "ledger") return;
    let stopped = false;
    void (async () => {
      try {
        const data = await api.ledger();
        if (!stopped) {
          setLedger(data);
          setLedgerError(null);
        }
      } catch (err) {
        if (!stopped) setLedgerError(err instanceof ApiError ? errorCopy(err.code) : errorCopy("unknown"));
      }
    })();
    return () => {
      stopped = true;
    };
  }, [api, health, tab]);

  async function onChoose(item: CatalogItem) {
    setSelected(item);
    setPicker(null);
    setError(null);
    setView("picker");
    setPickerLoading(true);
    try {
      const response = await api.picker(item.id);
      if (!mountedRef.current) return;
      setPicker(response);
    } catch (err) {
      if (!mountedRef.current) return;
      setError(err instanceof ApiError ? errorCopy(err.code) : errorCopy("unknown"));
    } finally {
      if (mountedRef.current) setPickerLoading(false);
    }
  }

  async function onPay(option: PickerOption) {
    if (!selected) return;
    setBusy(true);
    setError(null);
    try {
      const body = {
        itemId: selected.id,
        tokenAddress: option.tokenAddress,
        mode,
        ...(mode === "live" ? { passcode: passcode ?? "" } : {}),
      };
      const { checkoutId } = await api.startCheckout(body);
      storage.setItem(STORAGE_KEY, checkoutId);
      setEndedMessage(false);
      setRecorded(undefined);
      setCheckoutState(null);
      setView("progress");
      beginFollow(checkoutId);
    } catch (err) {
      if (!mountedRef.current) return;
      setError(err instanceof ApiError ? errorCopy(err.code) : errorCopy("unknown"));
    } finally {
      if (mountedRef.current) setBusy(false);
    }
  }

  function startOver() {
    sessionRef.current += 1;
    const handle = followRef.current;
    followRef.current = null;
    handle?.close();
    storage.removeItem(STORAGE_KEY);
    setCheckoutState(null);
    setEndedMessage(false);
    setRecorded(undefined);
    setSelected(null);
    setPicker(null);
    setPickerLoading(false);
    setError(null);
    setBusy(false);
    setView("catalog");
    setResumeId(null);
  }

  function onUnlocked(nextPasscode: string, nextBudget: BudgetResponse) {
    setPasscode(nextPasscode);
    setBudget(nextBudget);
    setMode("live");
    setUnlockOpen(false);
  }

  if (!health) return <BootScreen api={api} onReady={onReady} />;

  const showRestart = endedMessage;
  const showStartOver = view === "progress" || view === "missing" || view === "resuming";

  return (
    <div className="app-shell">
      <Header
        health={health}
        mode={mode}
        budget={budget}
        onSimulate={() => {
          setMode("simulate");
          setUnlockOpen(false);
        }}
        onRequestLive={() => setUnlockOpen(true)}
      />
      <div role="tablist" aria-label="Sections">
        <button type="button" role="tab" aria-selected={tab === "checkout"} onClick={() => setTab("checkout")}>
          Checkout
        </button>
        <button type="button" role="tab" aria-selected={tab === "ledger"} onClick={() => setTab("ledger")}>
          Ledger
        </button>
      </div>
      {unlockOpen && <UnlockDialog api={api} onUnlocked={onUnlocked} onCancel={() => setUnlockOpen(false)} />}
      {tab === "checkout" ? (
        <div role="tabpanel">
          {showRestart && <p>This checkout ended when the server restarted.</p>}
          {error && view !== "picker" && (
            <p role="alert" className="alert">
              {error}
            </p>
          )}
          {showStartOver && (
            <button type="button" onClick={startOver}>
              Start over
            </button>
          )}
          {view === "resuming" && <p>Loading checkout…</p>}
          {view === "progress" && checkoutState && (
            <>
              <Progress state={checkoutState} />
              {checkoutState.receipt && <ReceiptCard receipt={checkoutState.receipt} />}
              {checkoutState.deliverable && <DeliverableView deliverable={checkoutState.deliverable} />}
              {recorded !== undefined && checkoutState.status === "succeeded" && checkoutState.mode === "simulate" && (
                <RecordedRun receipt={recorded} />
              )}
            </>
          )}
          {view === "picker" && (
            <>
              <button
                type="button"
                className="link-button"
                onClick={() => {
                  setView("catalog");
                  setSelected(null);
                  setPicker(null);
                  setError(null);
                }}
              >
                ← Back to the catalog
              </button>
              {selected && <h2>{`${selected.title} · ${selected.merchantName}`}</h2>}
              {pickerLoading && <p>Checking your shares…</p>}
              {error && (
                <p role="alert" className="alert">
                  {error}
                </p>
              )}
              {picker && (
                <Picker response={picker} mode={mode} busy={busy} onPay={(option) => void onPay(option)} />
              )}
            </>
          )}
          {view === "catalog" && <Catalog items={items} onChoose={(item) => void onChoose(item)} />}
        </div>
      ) : (
        <div role="tabpanel">
          {ledgerError && (
            <p role="alert" className="alert">
              {ledgerError}
            </p>
          )}
          {ledger && <LedgerView ledger={ledger} />}
          <MerchantFeed api={api} />
        </div>
      )}
    </div>
  );
}
```

```ts
import { execSync } from "node:child_process";
import { existsSync, readFileSync, rmSync } from "node:fs";
import { join } from "node:path";
import { fileURLToPath } from "node:url";
import { describe, expect, it } from "vitest";

const root = fileURLToPath(new URL("../../../", import.meta.url));
const SECRETS = ["BINANCE_API_KEY", "BINANCE_API_SECRET", "WALLET_KEYSTORE_JSON", "WALLET_PASSWORD", "MERCHANT_PUBLIC_URL", "BUYER_PRIVATE_KEY", "LIVE_PASSCODE", "MERCHANT_URL"];

interface Service {
  name: string;
  fields: Record<string, string>;
  env: Array<{ key: string; value: string | null; sync: string | null }>;
}

function parseBlueprint(text: string): Service[] {
  const services: Service[] = [];
  let current: Service | null = null;
  let envEntry: Service["env"][number] | null = null;
  for (const raw of text.split(/\r?\n/)) {
    if (raw.trim().startsWith("#") || raw.trim() === "") continue;
    const service = raw.match(/^ {2}- type: (\S+)$/);
    if (service) {
      current = { name: "", fields: { type: service[1] ?? "" }, env: [] };
      services.push(current);
      envEntry = null;
      continue;
    }
    const field = raw.match(/^ {4}(\w+): ?(.*)$/);
    if (current && field) {
      current.fields[field[1] ?? ""] = (field[2] ?? "").trim();
      if (field[1] === "name") current.name = (field[2] ?? "").trim();
      continue;
    }
    const key = raw.match(/^ {6}- key: (\S+)$/);
    if (current && key) {
      envEntry = { key: key[1] ?? "", value: null, sync: null };
      current.env.push(envEntry);
      continue;
    }
    const attr = raw.match(/^ {8}(value|sync): (.*)$/);
    if (envEntry && attr) envEntry[attr[1] === "value" ? "value" : "sync"] = (attr[2] ?? "").replace(/^"(.*)"$/, "$1");
  }
  return services;
}

describe("render.yaml", () => {
  const services = parseBlueprint(readFileSync(join(root, "render.yaml"), "utf8"));
  const byName = new Map(services.map((s) => [s.name, s]));

  it("defines the two free Singapore web services", () => {
    expect(services.map((s) => s.name)).toEqual(["sharepay-merchant", "sharepay-checkout"]);
    for (const service of services) {
      expect(service.fields).toMatchObject({ type: "web", runtime: "node", plan: "free", region: "singapore" });
    }
    expect(byName.get("sharepay-merchant")?.fields).toMatchObject({ buildCommand: "npm ci", startCommand: "npm run start:merchant", healthCheckPath: "/ping" });
    expect(byName.get("sharepay-checkout")?.fields).toMatchObject({ buildCommand: "npm ci && npm run build:web", startCommand: "npm run start:checkout", healthCheckPath: "/api/health" });
  });

  it("asks for every secret in the dashboard and commits no value for it", () => {
    for (const service of services) {
      for (const entry of service.env) {
        if (SECRETS.includes(entry.key)) expect(entry, `${service.name} ${entry.key}`).toEqual({ key: entry.key, value: null, sync: "false" });
        else expect(entry.value, `${service.name} ${entry.key}`).not.toBeNull();
        expect(entry.value ?? "").not.toMatch(/0x[0-9a-fA-F]{64}/);
      }
    }
    const keys = (name: string) => byName.get(name)?.env.map((e) => e.key) ?? [];
    expect(keys("sharepay-merchant")).toEqual(expect.arrayContaining(["BINANCE_API_KEY", "BINANCE_API_SECRET", "WALLET_KEYSTORE_JSON", "WALLET_PASSWORD", "MERCHANT_PUBLIC_URL", "SWEEP_ENABLED"]));
    expect(keys("sharepay-checkout")).toEqual(expect.arrayContaining(["BINANCE_API_KEY", "BINANCE_API_SECRET", "BUYER_PRIVATE_KEY", "LIVE_PASSCODE", "MERCHANT_URL", "WALLET_MODE", "WEB_DIST"]));
    expect(byName.get("sharepay-checkout")?.env.find((e) => e.key === "WALLET_MODE")?.value).toBe("api-signer");
  });

  it("pins Node through .node-version and builds the production web app", () => {
    expect(readFileSync(join(root, ".node-version"), "utf8").trim()).toBe("22.21.0");
    const dist = join(root, "packages", "web", "dist");
    rmSync(dist, { recursive: true, force: true });
    const output = execSync("npm run build:web", { cwd: root, encoding: "utf8", stdio: "pipe" });
    expect(output).toContain("built in");
    expect(existsSync(join(dist, "index.html"))).toBe(true);
  }, 180_000);
});
```
