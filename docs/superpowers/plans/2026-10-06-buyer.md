# Buyer Engine Implementation Plan

**Goal:** Build `@sharepay/buyer`, the "pay with shares" engine. It includes:
- A market-status classifier and a SWAP-route probe.
- Holdings, measured in shares.
- The picker engine, which greys out stocks with reasons and ranks the rest.
- A sell planner that sells just enough shares.
- An x402 client and two payer wallets: the API signer (viem + Binance Trading/Transaction API) and the Agentic Wallet (`baw` CLI).
- Adapters for the three real merchants, a receipts store, and the checkout orchestrator that emits the 8-step state.

**Architecture:**
- Pure decision logic (`status`, `picker`, `planner` sizing) is separated from I/O (`routeProbe`, `holdings`, wallets, adapters).
- The I/O modules take injected clients, so every gate runs offline with fakes, throwaway keys and real captured fixtures.
- `checkout.ts` composes everything into one `runCheckout()` that streams `CheckoutState` snapshots.

**Tech Stack:** Node.js >= 22.12, TypeScript ~5.9.3, viem ^2.57.3, zod ^3.25.76, vitest ^3.2.7

**Source spec:** docs/superpowers/specs/2026-10-06-sharepay-design.md

**Conventions for every task in this plan:**
- Relative imports use `.js`; workspace imports use subpaths (`@sharepay/core/tokens`, `@sharepay/core/x402/codec`, …).
- No test touches the network or a real key.
- Gate: `npm run check -- <test file>`, which runs whole-repo `tsc` and then that vitest file.

import type { StatusInfo } from "@sharepay/core/binance/rwa";

export type StatusClass = "open" | "soft" | "hard";

export interface StatusVerdict {
  cls: StatusClass;
  code: string;
  message: string;
  nextOpenTime: number | null;
}

const KNOWN_MARKET_STATUSES = new Set(["premarket", "regular", "postmarket", "overnight", "closed", "pause"]);
const KNOWN_REASON_CODES = new Set(["TRADING", "MARKET_CLOSED", "MARKET_PAUSED", "MARKET_MAINTENANCE", "ASSET_PAUSED", "ASSET_LIMITED", "UNSUPPORTED"]);
const PAUSE_REASONS: Record<string, string> = {
  cash_dividend: "Paused: cash dividend",
  stock_dividend: "Paused: stock dividend",
  stock_split: "Paused: stock split",
  merger: "Paused: merger",
  acquisition: "Paused: acquisition",
  spinoff: "Paused: spin-off",
  maintenance: "Paused: maintenance",
  "corporate action": "Paused: corporate action",
};
const WEEKDAYS = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];

export function formatUtcShort(ms: number): string {
  const date = new Date(ms);
  const hh = String(date.getUTCHours()).padStart(2, "0");
  const mm = String(date.getUTCMinutes()).padStart(2, "0");
  return `${WEEKDAYS[date.getUTCDay()]} ${hh}:${mm} UTC`;
}

function verdict(cls: StatusClass, code: string, message: string, nextOpenTime: number | null = null): StatusVerdict {
  return { cls, code, message, nextOpenTime };
}

export function classifyStatus(status: StatusInfo | null | undefined): StatusVerdict {
  if (!status) return verdict("hard", "status_unavailable", "Market status unavailable");
  const reasonCode = status.reasonCode ?? null;
  const marketStatus = status.marketStatus ?? null;
  const open = status.openState === true;
  const next = typeof status.nextOpenTime === "number" && status.nextOpenTime > 0 ? status.nextOpenTime : null;
  const reopens = next ? ` · reopens ${formatUtcShort(next)}` : "";

  if (reasonCode !== null && !KNOWN_REASON_CODES.has(reasonCode)) {
    return verdict("hard", "unknown_status", `Unknown market status: ${reasonCode}`);
  }
  if (reasonCode === "ASSET_PAUSED") {
    const why = status.reasonMsg ?? "corporate action";
    return verdict("hard", "asset_paused", PAUSE_REASONS[why] ?? `Paused: ${why}`);
  }
  if (reasonCode === "MARKET_PAUSED" || marketStatus === "pause") return verdict("hard", "halted", "Trading halted");
  if (reasonCode === "MARKET_MAINTENANCE") return verdict("hard", "maintenance", "Market maintenance");
  if (reasonCode === "UNSUPPORTED") return verdict("hard", "unsupported", "Not supported for trading");
  if (reasonCode === "ASSET_LIMITED") return verdict("soft", "limited", "Earnings window: limited trading", next);
  if (reasonCode === "MARKET_CLOSED" || marketStatus === "closed") {
    return verdict("soft", "closed", `Market closed (${status.reasonMsg ?? "Weekend or Holiday"})${reopens}`, next);
  }
  if (marketStatus !== null && !KNOWN_MARKET_STATUSES.has(marketStatus)) {
    if (open && (reasonCode === "TRADING" || reasonCode === null)) return verdict("soft", "off_hours", "Off-hours session", next);
    return verdict("hard", "unknown_status", `Unknown market status: ${marketStatus}`);
  }
  if (open) return verdict("open", "open", `Open (${marketStatus ?? "trading"})`);
  return verdict("soft", "closed", `Market closed${reopens}`, next);
}

import { BinanceApiError } from "@sharepay/core/binance/client";
import { bestSwapRoute, hasRfqRoute, type QuoteRoute, type TradingApi } from "@sharepay/core/binance/trading";
import { getStable } from "@sharepay/core/tokens";
import type { Address } from "viem";

export interface SwapLeg {
  from: Address;
  to: Address;
  amountIn: bigint;
  route: QuoteRoute;
}

export type FailedRouteKind = "rfq_only" | "closed" | "below_min" | "above_max" | "no_liquidity" | "error";

export type RouteOutcome =
  | { kind: "swap"; legs: SwapLeg[]; expectedOut: bigint; priceImpactPercent: string | null }
  | { kind: FailedRouteKind; code: number | null; message: string; minUsd: string | null };

type FailedOutcome = { kind: FailedRouteKind; code: number | null; message: string; minUsd: string | null };

export interface ProbeParams {
  stock: Address;
  payAsset: Address;
  sellAmount: bigint;
  wallet: Address;
}

export function parseMinUsd(message: string): string | null {
  const match = message.match(/(\d+(?:\.\d+)?)/);
  return match?.[1] ?? null;
}

export function classifyQuoteError(error: unknown): FailedOutcome {
  if (error instanceof BinanceApiError) {
    const code = error.code;
    const message = `${code ?? `http ${error.httpStatus}`} ${error.apiMsg}`.trim();
    switch (code) {
      case 40367:
      case 40369:
        return { kind: "closed", code, message, minUsd: null };
      case 40375:
        return { kind: "below_min", code, message, minUsd: parseMinUsd(error.apiMsg) };
      case 40366:
        return { kind: "above_max", code, message, minUsd: null };
      case 40374:
        return { kind: "no_liquidity", code, message, minUsd: null };
      default:
        return { kind: "error", code, message, minUsd: null };
    }
  }
  return { kind: "error", code: null, message: error instanceof Error ? error.message : String(error), minUsd: null };
}

function maxImpact(a: string | null | undefined, b: string | null | undefined): string | null {
  const values = [a, b].filter((v): v is string => typeof v === "string").map(Number).filter((v) => Number.isFinite(v));
  return values.length > 0 ? String(Math.max(...values)) : null;
}

export async function probeRoute(trading: TradingApi, params: ProbeParams): Promise<RouteOutcome> {
  const failures: FailedOutcome[] = [];
  let rfqSeen = false;
  const attempt = async (from: Address, to: Address, amount: bigint, vendor?: "LiquidMesh"): Promise<QuoteRoute | null> => {
    try {
      const routes = await trading.quote({ amount, fromTokenAddress: from, toTokenAddress: to, userWalletAddress: params.wallet, vendor });
      if (hasRfqRoute(routes)) rfqSeen = true;
      return bestSwapRoute(routes);
    } catch (error) {
      failures.push(classifyQuoteError(error));
      return null;
    }
  };

  const direct =
    (await attempt(params.stock, params.payAsset, params.sellAmount)) ??
    (await attempt(params.stock, params.payAsset, params.sellAmount, "LiquidMesh"));
  if (direct) {
    return {
      kind: "swap",
      legs: [{ from: params.stock, to: params.payAsset, amountIn: params.sellAmount, route: direct }],
      expectedOut: BigInt(direct.toTokenAmount),
      priceImpactPercent: direct.priceImpactPercent ?? null,
    };
  }

  const usdt = getStable("USDT").address;
  const first = await attempt(params.stock, usdt, params.sellAmount, "LiquidMesh");
  if (first) {
    const usdtOut = BigInt(first.toTokenAmount);
    const second = await attempt(usdt, params.payAsset, usdtOut, "LiquidMesh");
    if (second) {
      return {
        kind: "swap",
        legs: [
          { from: params.stock, to: usdt, amountIn: params.sellAmount, route: first },
          { from: usdt, to: params.payAsset, amountIn: usdtOut, route: second },
        ],
        expectedOut: BigInt(second.toTokenAmount),
        priceImpactPercent: maxImpact(first.priceImpactPercent, second.priceImpactPercent),
      };
    }
  }

  const specific = failures.find((f) => f.kind === "closed" || f.kind === "below_min" || f.kind === "above_max");
  if (specific) return specific;
  if (rfqSeen) return { kind: "rfq_only", code: null, message: "Only an RFQ route is available", minUsd: null };
  return (
    failures.find((f) => f.kind === "no_liquidity") ??
    failures[0] ?? { kind: "no_liquidity", code: null, message: "No live route right now", minUsd: null }
  );
}

import type { RwaApi, RwaToken, StatusInfo } from "@sharepay/core/binance/rwa";
import type { WalletApi } from "@sharepay/core/binance/wallet";
import type { ChainReader } from "@sharepay/core/chain/rpc";
import { ONE, toUnits } from "@sharepay/core/money";
import { STOCKS, type StockToken } from "@sharepay/core/tokens";
import type { Address } from "viem";

export interface Holding {
  stock: StockToken;
  balance: bigint;
  priceUsd: string | null;
  sharesPerToken: bigint;
  status: StatusInfo | null;
}

export interface MarketSnapshot {
  tokens: Map<string, RwaToken>;
  at: number;
}

export async function loadMarketSnapshot(rwa: RwaApi, nowMs: number): Promise<MarketSnapshot> {
  const list = await rwa.tokens();
  const tokens = new Map<string, RwaToken>();
  for (const token of list) {
    if (token.binanceChainId === "56") tokens.set(token.tokenContractAddress.toLowerCase(), token);
  }
  return { tokens, at: nowMs };
}

export function ratioToMultiplier(ratio: string | null | undefined): bigint {
  if (!ratio) return ONE;
  try {
    const value = toUnits(ratio, 18);
    return value > 0n ? value : ONE;
  } catch {
    return ONE;
  }
}

export async function loadHoldings(
  deps: { wallet: WalletApi; chain: Pick<ChainReader, "balanceOf" | "uiMultiplier"> },
  owner: Address,
  snapshot: MarketSnapshot,
  stocks: readonly StockToken[] = STOCKS,
): Promise<Holding[]> {
  const balances = new Map<string, bigint>();
  try {
    const assets = await deps.wallet.tokenBalances(owner, stocks.map((s) => s.address));
    for (const asset of assets) {
      if (asset.rawBalance && /^\d+$/.test(asset.rawBalance)) {
        balances.set(asset.tokenContractAddress.toLowerCase(), BigInt(asset.rawBalance));
      }
    }
  } catch {
    for (const stock of stocks) {
      balances.set(stock.address, await deps.chain.balanceOf(stock.address, owner));
    }
  }
  const holdings: Holding[] = [];
  for (const stock of stocks) {
    const balance = balances.get(stock.address) ?? 0n;
    if (balance === 0n) continue;
    const token = snapshot.tokens.get(stock.address) ?? null;
    let sharesPerToken = ratioToMultiplier(token?.tokenToShareRatio);
    if (stock.platform === "bstock") {
      try {
        sharesPerToken = await deps.chain.uiMultiplier(stock.address);
      } catch {
        // keep the RWA tokenToShareRatio
      }
    }
    holdings.push({ stock, balance, priceUsd: token?.tokenPrice ?? null, sharesPerToken, status: token?.statusInfo ?? null });
  }
  return holdings;
}

import type { TradingApi } from "@sharepay/core/binance/trading";
import { ONE, addBps, mulDivUp, sharesFromTokens, toUnits } from "@sharepay/core/money";
import type { StockToken } from "@sharepay/core/tokens";
import type { Address } from "viem";
import { probeRoute, type SwapLeg } from "./routeProbe.js";

export interface SaleEstimate {
  sellAmount: bigint;
  sellShares: bigint;
}

export interface SalePlan {
  stock: StockToken;
  payAsset: Address;
  priceAtomic: bigint;
  legs: SwapLeg[];
  sellAmount: bigint;
  expectedOut: bigint;
  priceImpactPercent: string | null;
}

export class PlanError extends Error {
  readonly code: "no_price" | "quote_short" | "insufficient_shares" | "route_lost";

  constructor(code: "no_price" | "quote_short" | "insufficient_shares" | "route_lost", message: string) {
    super(message);
    this.name = "PlanError";
    this.code = code;
  }
}

export const SALE_BUFFER_BPS = 300;
export const QUOTE_MARGIN_BPS = 150;
export const TWO_LEG_MARGIN_BPS = 300;

function targetFor(priceAtomic: bigint, legs: number): bigint {
  return addBps(priceAtomic, legs === 2 ? TWO_LEG_MARGIN_BPS : QUOTE_MARGIN_BPS);
}

export function estimateSale(params: { priceAtomic: bigint; tokenPriceUsd: string | null; sharesPerToken: bigint }): SaleEstimate {
  if (!params.tokenPriceUsd) throw new PlanError("no_price", "no live token price");
  const price = toUnits(params.tokenPriceUsd, 18);
  if (price === 0n) throw new PlanError("no_price", "token price is zero");
  const sellAmount = addBps(mulDivUp(params.priceAtomic, ONE, price), SALE_BUFFER_BPS);
  return { sellAmount, sellShares: sharesFromTokens(sellAmount, params.sharesPerToken) };
}

export async function planSale(
  trading: TradingApi,
  params: {
    stock: StockToken;
    payAsset: Address;
    priceAtomic: bigint;
    estimate: SaleEstimate;
    outcome: { kind: "swap"; legs: SwapLeg[]; expectedOut: bigint; priceImpactPercent: string | null };
    wallet: Address;
    balance: bigint;
  },
): Promise<SalePlan> {
  let sellAmount = params.estimate.sellAmount;
  let outcome = params.outcome;
  if (sellAmount > params.balance) {
    throw new PlanError("insufficient_shares", `need ${sellAmount} tokens, hold ${params.balance}`);
  }
  const target = targetFor(params.priceAtomic, outcome.legs.length);
  if (outcome.expectedOut < target) {
    if (outcome.expectedOut === 0n) throw new PlanError("quote_short", "route returns nothing");
    sellAmount = addBps(mulDivUp(sellAmount, target, outcome.expectedOut), 100);
    if (sellAmount > params.balance) {
      throw new PlanError("insufficient_shares", `need ${sellAmount} tokens after resizing, hold ${params.balance}`);
    }
    const again = await probeRoute(trading, { stock: params.stock.address, payAsset: params.payAsset, sellAmount, wallet: params.wallet });
    if (again.kind !== "swap") throw new PlanError("route_lost", `route disappeared while resizing: ${again.message}`);
    const againTarget = targetFor(params.priceAtomic, again.legs.length);
    if (again.expectedOut < againTarget) throw new PlanError("quote_short", `quote ${again.expectedOut} is still below ${againTarget}`);
    outcome = again;
  }
  return {
    stock: params.stock,
    payAsset: params.payAsset,
    priceAtomic: params.priceAtomic,
    legs: outcome.legs,
    sellAmount,
    expectedOut: outcome.expectedOut,
    priceImpactPercent: outcome.priceImpactPercent,
  };
}

import type { Badge, PickerOption } from "@sharepay/core/dto";
import { formatAmount, formatSignificant, sharesFromTokens, usdValue } from "@sharepay/core/money";
import type { Address } from "viem";
import type { Holding } from "./holdings.js";
import { PlanError, estimateSale, type SaleEstimate } from "./planner.js";
import type { ProbeParams, RouteOutcome } from "./routeProbe.js";
import { classifyStatus, formatUtcShort, type StatusClass, type StatusVerdict } from "./status.js";

export interface OptionInput {
  holding: Holding;
  status: StatusVerdict;
  estimate: SaleEstimate | null;
  outcome: RouteOutcome | null;
}

export function evaluateOption(input: OptionInput): PickerOption {
  const { holding, status, estimate, outcome } = input;
  const sharesHeld = formatSignificant(sharesFromTokens(holding.balance, holding.sharesPerToken), 18, 4);
  const base = {
    tokenAddress: holding.stock.address,
    symbol: holding.stock.symbol,
    name: holding.stock.name,
    platform: holding.stock.platform,
    sharesHeld,
    valueUsd: holding.priceUsd ? usdValue(holding.balance, holding.priceUsd) : "0",
    rank: null,
  };
  const grey = (code: string, message: string): PickerOption => ({ ...base, payable: false, badges: [], reason: { code, message }, plan: null });

  if (status.cls === "hard") return grey(status.code, status.message);
  if (!estimate) return grey("no_price", "No live price for this stock right now");
  if (holding.balance < estimate.sellAmount) {
    return grey("insufficient_shares", `Not enough shares: need ${formatSignificant(estimate.sellShares, 18, 4)}, have ${sharesHeld}`);
  }
  if (!outcome) return grey("no_route", "No live route right now");
  switch (outcome.kind) {
    case "swap": {
      const badges: Badge[] = status.cls === "soft" ? [{ tone: "amber", text: status.message }] : [];
      return {
        ...base,
        payable: true,
        badges,
        reason: null,
        plan: {
          sellTokens: formatSignificant(estimate.sellAmount, 18, 4),
          sellShares: formatSignificant(estimate.sellShares, 18, 4),
          expectedOut: formatAmount(outcome.expectedOut, 18, 4),
          priceImpactPercent: outcome.priceImpactPercent,
          legs: outcome.legs.length === 2 ? 2 : 1,
        },
      };
    }
    case "rfq_only":
      return grey("rfq_only", "Only an RFQ route is available (not used for sub-dollar payments)");
    case "below_min":
      return grey("below_min", outcome.minUsd ? `Below Ondo's minimum order ($${outcome.minUsd})` : "Below the issuer's minimum order");
    case "above_max":
      return grey("above_max", "Above the maximum single order");
    case "closed":
      return grey("closed_no_route", `Market closed and no 24/7 route${status.nextOpenTime ? ` · reopens ${formatUtcShort(status.nextOpenTime)}` : ""}`);
    case "no_liquidity":
      return grey("no_route", "No live route right now");
    case "error":
      return grey("route_error", `No live route (${outcome.message})`);
  }
}

function impactOf(option: PickerOption): number {
  const value = Number(option.plan?.priceImpactPercent ?? "100");
  return Number.isFinite(value) ? value : 100;
}

export function rankOptions(evaluated: Array<{ option: PickerOption; statusCls: StatusClass }>): PickerOption[] {
  const payable = evaluated
    .filter((e) => e.option.payable)
    .sort(
      (a, b) =>
        (a.statusCls === "open" ? 0 : 1) - (b.statusCls === "open" ? 0 : 1) ||
        impactOf(a.option) - impactOf(b.option) ||
        Number(b.option.valueUsd) - Number(a.option.valueUsd) ||
        a.option.symbol.localeCompare(b.option.symbol),
    );
  const greyed = evaluated.filter((e) => !e.option.payable).sort((a, b) => a.option.symbol.localeCompare(b.option.symbol));
  return [...payable.map((e, i) => ({ ...e.option, rank: i + 1 })), ...greyed.map((e) => ({ ...e.option, rank: null }))];
}

export async function computePicker(
  deps: { probe: (params: ProbeParams) => Promise<RouteOutcome> },
  params: { holdings: Holding[]; priceAtomic: bigint; payAsset: Address; wallet: Address },
): Promise<PickerOption[]> {
  const evaluated: Array<{ option: PickerOption; statusCls: StatusClass }> = [];
  for (const holding of params.holdings) {
    const status = classifyStatus(holding.status);
    let estimate: SaleEstimate | null = null;
    let outcome: RouteOutcome | null = null;
    if (status.cls !== "hard") {
      try {
        estimate = estimateSale({ priceAtomic: params.priceAtomic, tokenPriceUsd: holding.priceUsd, sharesPerToken: holding.sharesPerToken });
      } catch (error) {
        if (!(error instanceof PlanError)) throw error;
      }
      if (estimate && holding.balance >= estimate.sellAmount) {
        outcome = await deps.probe({ stock: holding.stock.address, payAsset: params.payAsset, sellAmount: estimate.sellAmount, wallet: params.wallet });
      }
    }
    evaluated.push({ option: evaluateOption({ holding, status, estimate, outcome }), statusCls: status.cls });
  }
  return rankOptions(evaluated);
}

import type { ParsedChallenge, SelectedAccept, SettlementResponse } from "@sharepay/core/x402/types";

export interface PaidRequest {
  url: string;
  init: RequestInit;
  timeoutMs: number;
}

export class X402ClientError extends Error {
  readonly code: "expected_402" | "merchant_error";

  constructor(code: "expected_402" | "merchant_error", message: string) {
    super(message);
    this.name = "X402ClientError";
    this.code = code;
  }
}

async function readBody(response: Response): Promise<unknown> {
  const text = await response.text();
  try {
    return text.length > 0 ? JSON.parse(text) : null;
  } catch {
    return null;
  }
}

export async function fetchChallenge(
  fetchImpl: typeof fetch,
  request: PaidRequest,
  options: { maxAmount: bigint; preferredAssets: string[] },
): Promise<{ challenge: ParsedChallenge; selected: SelectedAccept }> {
  const response = await fetchImpl(request.url, { ...request.init, signal: AbortSignal.timeout(request.timeoutMs) });
  if (response.status !== 402) {
    throw new X402ClientError("expected_402", `merchant answered HTTP ${response.status} instead of 402`);
  }
  const body = await readBody(response);
  const challenge = parseChallenge(response.headers.get("payment-required"), body);
  const selected = selectEip3009Accept(challenge, { network: X402_NETWORK, preferredAssets: options.preferredAssets, maxAmount: options.maxAmount });
  return { challenge, selected };
}

export async function replayWithPayment(
  fetchImpl: typeof fetch,
  request: PaidRequest,
  header: { name: string; value: string },
): Promise<{ response: Response; settlement: SettlementResponse | null }> {
  const headers = new Headers(request.init.headers);
  headers.set(header.name, header.value);
  const response = await fetchImpl(request.url, { ...request.init, headers, signal: AbortSignal.timeout(request.timeoutMs) });
  return { response, settlement: decodeSettlementResponse(response.headers.get("payment-response")) };
}

export function signedHeaderWindow(headerValue: string, nowSec: number): { ok: boolean; reason: string | null } {
  let validAfter: bigint;
  let validBefore: bigint;
  try {
    const payload = decodePaymentPayload(headerValue);
    validAfter = BigInt(payload.payload.authorization.validAfter);
    validBefore = BigInt(payload.payload.authorization.validBefore);
  } catch (error) {
    return { ok: false, reason: `undecodable payment header: ${error instanceof Error ? error.message : String(error)}` };
  }
  const now = BigInt(nowSec);
  if (validAfter > now) return { ok: false, reason: "signature is not valid yet" };
  if (validBefore < now + 6n) return { ok: false, reason: "signature expires too soon" };
  if (validBefore - now > 3600n) return { ok: false, reason: "signature window is more than an hour" };
  return { ok: true, reason: null };
}

export async function freshSwapRoute(trading: TradingApi, leg: SwapLeg, amountIn: bigint, wallet: Address): Promise<QuoteRoute | null> {
  const vendor = (leg.route.vendorName ?? "") === "LiquidMesh" ? "LiquidMesh" : undefined;

export async function simulateFirstLeg(deps: SimDeps, owner: Address, plan: SalePlan): Promise<SaleSimulation> {
  const leg = plan.legs[0];
  if (!leg) throw new SaleError("no_route", "the sale plan has no legs");
  const spender = (leg.route.approveTarget ?? AGGREGATION_ROUTER) as Address;
  const allowance = await deps.chain.allowance(leg.from, owner, spender);
  if (allowance < leg.amountIn) {
    const approve = await deps.trading.approveTransaction({ tokenContractAddress: leg.from, approveAmount: MAX_UINT256, vendor: leg.route.vendorName ?? undefined });
    if (!sameAddress(approve.dexContractAddress, spender)) {
      return { ok: false, mode: "approve_only", failReason: `approve spender ${approve.dexContractAddress} does not match route spender ${spender}`, detail: {} };
    }
    const simulation = await deps.tx.simulate({ from: owner, to: leg.from, value: 0n, data: approve.data });
    return {
      ok: simulation.ok,
      mode: "approve_only",
      failReason: simulation.ok ? null : simulation.failReason ?? simulation.status,
      detail: { approveSpender: spender, note: "The swap can be simulated after the approval lands" },
    };
  }
  const route = await freshSwapRoute(deps.trading, leg, leg.amountIn, owner);
  if (!route) return { ok: false, mode: "swap", failReason: "no SWAP route right now", detail: {} };
  const build = await deps.trading.swap({
    amount: leg.amountIn,
    fromTokenAddress: leg.from,
    toTokenAddress: leg.to,
    userWalletAddress: owner,
    quoteId: route.quoteId,
    slippagePercent: deps.slippagePercent,
  });
  if ((build.executionMode ?? "").toUpperCase() !== "SWAP" || !build.tx) {
    return { ok: false, mode: "swap", failReason: "swap build is not a SWAP transaction", detail: {} };
  }
  const simulation = await deps.tx.simulate({ from: owner, to: build.tx.to, value: BigInt(build.tx.value ?? "0"), data: build.tx.data });
  const changes = simulation.balanceChanges

    .map((c) => `${c.contractAddress ?? "native"}:${c.change}`)
    .join(", ");
  return {
    ok: simulation.ok,
    mode: "swap",
    failReason: simulation.ok ? null : simulation.failReason ?? simulation.status,
    detail: { router: build.tx.to, minReceive: build.tx.minReceiveAmount ?? "", balanceChanges: changes, vendor: route.vendorName ?? "" },

            if (!approveSim.ok) throw new SaleError("simulation_failed", `approve simulation failed: ${approveSim.failReason ?? approveSim.status}`);
            onBroadcast?.();
            txHashes.push((await sendPrepared(pipeline, approval, { mevProtection: false, receiptTimeoutMs: deps.receiptTimeoutMs, gasFallback: 80_000n })).txHash);
            route = await freshSwapRoute(deps.trading, leg, amountIn, owner);
            if (!route) throw new SaleError("route_lost", `no SWAP route for leg ${i + 1} after approval`);
          }
          const build = await deps.trading.swap({
            amount: amountIn,
            fromTokenAddress: leg.from,
            toTokenAddress: leg.to,
            userWalletAddress: owner,
            quoteId: route.quoteId,
            slippagePercent: deps.slippagePercent,
          });
          if ((build.executionMode ?? "").toUpperCase() !== "SWAP" || !build.tx) {
            throw new SaleError("not_swap", "swap build is not a SWAP transaction");
          }
          if (last && BigInt(build.tx.minReceiveAmount ?? "0") < plan.priceAtomic) {
            throw new SaleError("price_moved", `minimum receive ${build.tx.minReceiveAmount ?? "0"} is below the price ${plan.priceAtomic}`);
          }
          const prepared = { to: build.tx.to as Address, data: build.tx.data as Hex, value: BigInt(build.tx.value ?? "0") };
          const simulation = await simulatePrepared(deps.tx, owner, prepared);
          if (!simulation.ok) throw new SaleError("simulation_failed", `swap simulation failed: ${simulation.failReason ?? simulation.status}`);
          const before = await deps.chain.balanceOf(leg.to, owner);
          onBroadcast?.();
          txHashes.push((await sendPrepared(pipeline, prepared, { mevProtection: true, receiptTimeoutMs: deps.receiptTimeoutMs, gasFallback: 450_000n })).txHash);
          const after = await deps.chain.balanceOf(leg.to, owner);
          received = after > before ? after - before : 0n;
          if (received === 0n) throw new SaleError("nothing_received", `leg ${i + 1} returned nothing`);
          amountIn = received;
        }
        if (received < plan.priceAtomic) {
          throw new SaleError("price_moved", `received ${received} is below the price ${plan.priceAtomic}`);
        }
        return { txHashes, received };
      }),
    async signPayment(challenge, selected, nowSec) {
      const authorization = buildAuthorization({
        from: owner,
        to: selected.accept.payTo as Address,
        value: BigInt(selected.accept.amount),
        nowSec,
        maxTimeoutSeconds: selected.accept.maxTimeoutSeconds ?? 300,
      });
      const signature = await viemAuthorizationSigner(deps.account).signTransferAuthorization(domainForStable(selected.stable), authorization);
      return {
        name: HEADER_PAYMENT_SIGNATURE,
        value: encodePaymentPayload({ x402Version: 2, resource: challenge.required.resource, accepted: selected.accept, payload: { signature, authorization } }),
      };
    },
  };
}

import { execFile } from "node:child_process";

export type ExecFn = (file: string, args: string[], options: { timeoutMs: number }) => Promise<{ stdout: string; stderr: string; code: number }>;

export const nodeExec: ExecFn = (file, args, options) =>
  new Promise((resolve) => {
    execFile(file, args, { timeout: options.timeoutMs, maxBuffer: 10 * 1024 * 1024, windowsHide: true }, (error, stdout, stderr) => {
      resolve({ stdout: String(stdout), stderr: String(stderr), code: error ? 1 : 0 });
    });
  });

export function bawCommand(bin: string): { file: string; prefix: string[] } {
  return /\.(c|m)?js$/i.test(bin) ? { file: process.execPath, prefix: [bin] } : { file: bin, prefix: [] };
}

export class BawError extends Error {
  readonly command: string;

  constructor(command: string, message: string) {
    super(`baw ${command}: ${message}`);
    this.name = "BawError";
    this.command = command;
  }
}

export interface BawPaymentOption {
  index: number;
  status: string;
  reasons: string[];
  originalAccept: Record<string, unknown> | null;
}

export interface BawClient {
  address(): Promise<Address>;
  walletStatus(): Promise<string>;
  swap(params: { fromToken: Address; toToken: Address; fromTokenQty: string }): Promise<string>;
  order(orderId: string): Promise<{ status: string; txHash: string | null }>;
  x402Preview(paymentRequirements: string): Promise<{ paymentId: string; options: BawPaymentOption[] }>;
  x402Sign(paymentId: string, selectedIndex: number): Promise<{ headerName: string; headerValue: string; signatureExpiresAt: number | null }>;
}

function asRecord(value: unknown): Record<string, unknown> {
  return value !== null && typeof value === "object" ? (value as Record<string, unknown>) : {};
}

export function createBawClient(exec: ExecFn, bin: string): BawClient {
  async function run(args: string[], timeoutMs = 60_000): Promise<Record<string, unknown>> {
    const { file, prefix } = bawCommand(bin);
    const command = args.slice(0, 2).join(" ");
    const result = await exec(file, [...prefix, ...args], { timeoutMs });
    const text = result.stdout.trim();
    const start = text.indexOf("{");
    let parsed: unknown;
    try {
      parsed = JSON.parse(start >= 0 ? text.slice(start) : text);
    } catch {
      throw new BawError(command, `non-JSON output: ${(result.stderr || result.stdout).slice(0, 300)}`);
    }
    const record = asRecord(parsed);
    if (record.success !== true) {
      const why = record.error ?? record.message ?? record.msg ?? JSON.stringify(record).slice(0, 300);
      throw new BawError(command, String(why));
    }
    return asRecord(record.data);
  }

  return {
    async address() {
      const data = await run(["wallet", "address", "--json"]);
      const list = Array.isArray(data.addresses) ? data.addresses.map(asRecord) : [];
      const bsc = list.find((entry) => String(entry.binanceChainId) === "56");
      if (!bsc || typeof bsc.address !== "string") throw new BawError("wallet address", "no BSC (56) address");
      return bsc.address as Address;
    },
    async walletStatus() {
      return String((await run(["wallet", "status", "--json"])).status ?? "UNKNOWN");
    },
    async swap(params) {
      const data = await run(
        ["market-order", "swap", "--fromTokenQty", params.fromTokenQty, "--fromToken", params.fromToken, "--toToken", params.toToken, "--binanceChainId", "56", "--slippage", "auto", "--mev", "true", "--json"],
        120_000,
      );
      if (typeof data.orderId !== "string" && typeof data.orderId !== "number") throw new BawError("market-order swap", "no orderId");
      return String(data.orderId);
    },
    async order(orderId) {
      const data = await run(["market-order", "list", "--orderId", orderId, "--json"]);
      const list = Array.isArray(data.list) ? data.list.map(asRecord) : [];
      const entry = list.find((o) => String(o.orderId) === orderId) ?? list[0];
      if (!entry) return { status: "PENDING", txHash: null };
      return { status: String(entry.status ?? "PENDING"), txHash: typeof entry.txHash === "string" ? entry.txHash : null };
    },
    async x402Preview(paymentRequirements) {
      const data = await run(["x402-payment", "preview", "--paymentRequirements", paymentRequirements, "--json"]);
      const options = (Array.isArray(data.options) ? data.options : []).map(asRecord).map((o) => ({
        index: Number(o.index),
        status: String(o.status ?? ""),
        reasons: Array.isArray(o.reasons) ? o.reasons.map(String) : [],
        originalAccept: o.originalAccept !== null && typeof o.originalAccept === "object" ? (o.originalAccept as Record<string, unknown>) : null,
      }));
      return { paymentId: String(data.paymentId ?? ""), options };
    },
    async x402Sign(paymentId, selectedIndex) {
      const data = await run(["x402-payment", "sign", "--paymentId", paymentId, "--selectedIndex", String(selectedIndex), "--json"]);
      return {
        headerName: String(data.paymentHeaderName ?? "PAYMENT-SIGNATURE"),
        headerValue: String(data.paymentHeaderValue ?? ""),
        signatureExpiresAt: typeof data.signatureExpiresAt === "number" ? data.signatureExpiresAt : null,
      };
    },
  };
}

export function matchOption(options: BawPaymentOption[], selected: SelectedAccept): BawPaymentOption | null {
  const target = canonicalJson(selected.accept);
  const exact = options.find((o) => o.originalAccept !== null && canonicalJson(o.originalAccept) === target);
  if (exact) return exact;
  return (
    options.find((o) => {
      const original = o.originalAccept ?? {};
      const extra = asRecord(original.extra);
      const method = typeof extra.assetTransferMethod === "string" ? extra.assetTransferMethod : "eip3009";
      return typeof original.asset === "string" && sameAddress(original.asset, selected.accept.asset) && method === "eip3009" && String(original.amount ?? "") === selected.accept.amount;
    }) ?? null
  );
}

export function createAgenticWallet(deps: AgenticWalletDeps): PayerWallet {
  const pollMs = deps.pollMs ?? 3000;
  const swapTimeoutMs = deps.swapTimeoutMs ?? 120_000;
  let cached: Address | null = null;
  const address = async (): Promise<Address> => {
    if (!cached) cached = await deps.baw.address();
    return cached;
  };

    kind: "agentic-wallet",

      return simulateFirstLeg({ ...deps.sim, chain: deps.chain }, await address(), plan);
    },
    async executeSale(plan, onBroadcast) {
      const owner = await address();
      const txHashes: Hex[] = [];
      let amountIn = plan.sellAmount;
      let received = 0n;
      for (const leg of plan.legs) {
        const before = await deps.chain.balanceOf(leg.to, owner);
        onBroadcast?.();
        const orderId = await deps.baw.swap({ fromToken: leg.from, toToken: leg.to, fromTokenQty: formatUnits(amountIn, 18) });
        const started = deps.nowMs();
        let txHash: string | null = null;
        for (let polls = 0; ; polls += 1) {
          const order = await deps.baw.order(orderId);
          if (order.status === "FINISHED") {
            txHash = order.txHash;
            break;
          }
          if (order.status === "FAILED") throw new SaleError("aw_swap_failed", `Agentic Wallet order ${orderId} failed`);
          if (deps.nowMs() - started > swapTimeoutMs || polls * pollMs > swapTimeoutMs) {
            throw new SaleError("aw_swap_timeout", `Agentic Wallet order ${orderId} is still processing`);
          }
          await deps.sleep(pollMs);
        }
        if (txHash) txHashes.push(txHash as Hex);
        const after = await deps.chain.balanceOf(leg.to, owner);
        received = after > before ? after - before : 0n;
        if (received === 0n) throw new SaleError("nothing_received", `order ${orderId} returned nothing`);
        amountIn = received;
      }
      if (received < plan.priceAtomic) throw new SaleError("price_moved", `received ${received} is below the price ${plan.priceAtomic}`);
      return { txHashes, received };
    },
    async signPayment(challenge, selected, nowSec) {
      const requirements = challengeHeaderValue(challenge);
      for (let attempt = 1; attempt <= 5; attempt += 1) {
        const preview = await deps.baw.x402Preview(requirements);
        const option = matchOption(preview.options, selected);
        if (option?.status === "READY_TO_SIGN") {
          const signed = await deps.baw.x402Sign(preview.paymentId, option.index);
          const window = signedHeaderWindow(signed.headerValue, nowSec);
          if (!window.ok) throw new SaleError("aw_window", window.reason ?? "bad signature window");
          return { name: signed.headerName, value: signed.headerValue };
        }
        if (option?.status === "NOT_SIGNABLE") {
          throw new SaleError("aw_not_signable", `Agentic Wallet cannot sign: ${option.reasons.join(", ") || "no reason given"}`);
        }
        if (attempt < 5) await deps.sleep(3000);
      }
      throw new SaleError("aw_not_ready", "Agentic Wallet never offered a READY_TO_SIGN option for this payment");
    },
  };
}

  retryable?: boolean;

const DEADLINE_MS = 10 * 60 * 1000;

export function stockAnalystAdapter(baseUrl = "https://stock-agent.bnbchain.org"): MerchantAdapter {
  const base = baseUrl.replace(/\/+$/, "");
  return {
    id: "bnb-stock-analyst-nvda",
    title: "NVDA full analysis",
    merchantName: "BNB Chain stock analyst (Agent Studio)",
    merchantUrl: base,
    description: "BNB Chain's own Agent Studio stock analyst writes a full NVDA analysis; payment settles through Binance's B402 facilitator.",
    deliveryHint: "Arrives in about 2–5 min",
    kind: "async",
    etaSec: 240,
    request: () => ({
      url: `${base}/x402/analyze/async`,
      init: { method: "POST", headers: { Accept: "application/json", "Content-Type": "application/json" }, body: JSON.stringify({ symbols: ["NVDA"] }) },
      timeoutMs: 60_000,
    }),
    async deliver(response: Response, ctx: DeliveryContext) {
      const receipt = (await response.json().catch(() => ({}))) as { jobId?: string; jobToken?: string; error?: string };
      if ((response.status !== 202 && response.status !== 200) || !receipt.jobId || !receipt.jobToken) {
        throw new DeliveryError("merchant_error", `stock analyst answered HTTP ${response.status}: ${receipt.error ?? "no job receipt"}`);

      const jobStartedAt = new Date(startedAt).toISOString();

      for (;;) {

        const job = (await res.json().catch(() => ({}))) as JobStatus;
        const polledAt = new Date(ctx.nowMs()).toISOString();
        const elapsedSec = Math.round((ctx.nowMs() - startedAt) / 1000);
        ctx.onProgress({ etaSec: Math.max(240 - elapsedSec, 0), jobStartedAt, lastPolledAt: polledAt, note: `Report generating (${job.status ?? "unknown"})` });
        if (job.status === "succeeded" && job.downloadUrl) {
          const download = await ctx.fetchImpl(job.downloadUrl, { method: "GET", headers: { Accept: "text/markdown, text/plain;q=0.9" }, signal: AbortSignal.timeout(60_000) });
          if (!download.ok) throw new DeliveryError("download_failed", `report download failed with HTTP ${download.status}`);
          return { kind: "markdown", title: "NVDA full analysis (BNB Chain stock analyst)", markdown: await download.text() };
        }
        if (job.status === "failed") {
          if (job.retryable === true && !resumed) {
            resumed = true;
            await ctx.fetchImpl(`${jobUrl}/resume`, { method: "POST", headers, signal: AbortSignal.timeout(30_000) });
          } else {

          await ctx.sleep(Number.isFinite(retryAfter) && retryAfter * 1000 > POLL_MS ? retryAfter * 1000 : POLL_MS);
        }
      }
    },
  };

import { DeliveryError, type MerchantAdapter } from "./types.js";

export function parseSseJson(body: string): unknown {
  const dataLines = body
    .split(/\r?\n/)
    .filter((line) => line.startsWith("data:"))
    .map((line) => line.slice(5).trim())
    .filter((line) => line.length > 0);
  const text = dataLines.length > 0 ? dataLines[dataLines.length - 1] ?? "" : body.trim();
  return JSON.parse(text);
}

function asRecord(value: unknown): Record<string, unknown> {
  return value !== null && typeof value === "object" ? (value as Record<string, unknown>) : {};
}

export function cmcDossierAdapter(endpoint = "https://mcp.coinmarketcap.com/x402/mcp"): MerchantAdapter {
  return {
    id: "cmc-nvda-dossier",
    title: "NVDA research dossier",
    merchantName: "CoinMarketCap",
    merchantUrl: endpoint,
    description: "CoinMarketCap's us_equity_research_dossier skill: fundamentals, valuation, filings and risks with citations.",
    deliveryHint: "Arrives in about 1–5 min",
    kind: "async",
    etaSec: 180,
    request: () => ({
      url: endpoint,
      init: {
        method: "POST",
        headers: { "Content-Type": "application/json", Accept: "application/json, text/event-stream" },
        body: JSON.stringify({
          jsonrpc: "2.0",
          id: 1,
          method: "tools/call",
          params: { name: "execute_skill", arguments: { unique_name: "us_equity_research_dossier", parameters: { symbol: "NVDA" } } },
        }),
      },
      timeoutMs: 330_000,
    }),
    async deliver(response) {
      if (!response.ok) throw new DeliveryError("merchant_error", `CoinMarketCap answered HTTP ${response.status}`);
      let rpc: Record<string, unknown>;
      try {
        rpc = asRecord(parseSseJson(await response.text()));
      } catch {
        throw new DeliveryError("bad_response", "CoinMarketCap returned an unreadable response");
      }
      if (rpc.error) throw new DeliveryError("rpc_error", `CoinMarketCap error: ${JSON.stringify(rpc.error).slice(0, 200)}`);
      const result = asRecord(rpc.result);
      const content = Array.isArray(result.content) ? result.content.map(asRecord) : [];
      const text = typeof content[0]?.text === "string" ? content[0].text : "";
      if (result.isError === true) throw new DeliveryError("tool_error", `CoinMarketCap tool error: ${text.slice(0, 200)}`);
      let payload: Record<string, unknown>;
      try {
        payload = asRecord(JSON.parse(text));
      } catch {
        throw new DeliveryError("bad_response", `CoinMarketCap returned non-JSON content: ${text.slice(0, 120)}`);
      }
      const nested = asRecord(payload.result);
      const ok = payload.ok === true || nested.ok === true;
      if (!ok) throw new DeliveryError("skill_failed", `CoinMarketCap skill did not succeed: ${JSON.stringify(payload).slice(0, 200)}`);
      return { kind: "json", title: "NVDA research dossier (CoinMarketCap)", data: payload.data ?? nested.data ?? payload };
    },
  };
}

import { existsSync, mkdirSync, readFileSync, writeFileSync } from "node:fs";
import { dirname } from "node:path";
import type { ItemId, Receipt } from "@sharepay/core/dto";

export interface ReceiptStore {
  list(): Receipt[];
  add(receipt: Receipt): void;
  update(id: string, patch: Partial<Receipt>): void;
  latestFor(itemId: ItemId): Receipt | null;
}

function readList(path: string | null): Receipt[] {
  if (!path || !existsSync(path)) return [];
  try {
    const parsed = JSON.parse(readFileSync(path, "utf8")) as unknown;
    return Array.isArray(parsed) ? (parsed as Receipt[]) : [];
  } catch {
    return [];
  }
}

export async function createReceiptStore(options: {
  seedPath: string | null;
  runtimePath: string | null;
  verify?: (receipt: Receipt) => Promise<boolean>;
}): Promise<ReceiptStore> {
  const seed: Receipt[] = [];
  for (const candidate of readList(options.seedPath)) {
    let ok = true;
    if (options.verify) {
      try {
        ok = await options.verify(candidate);
      } catch {
        ok = false;
      }
    }
    if (ok) seed.push(candidate);
  }
  const seedIds = new Set(seed.map((r) => r.id));
  const items: Receipt[] = [...seed];
  for (const runtime of readList(options.runtimePath)) {
    if (!seedIds.has(runtime.id) && !items.some((r) => r.id === runtime.id)) items.push(runtime);
  }
  items.sort((a, b) => a.createdAt.localeCompare(b.createdAt));

  function persist(): void {
    if (!options.runtimePath) return;
    try {
      mkdirSync(dirname(options.runtimePath), { recursive: true });
      writeFileSync(options.runtimePath, JSON.stringify(items.filter((r) => !seedIds.has(r.id)), null, 2), "utf8");
    } catch {
      // best-effort on hosts without a persistent disk
    }
  }

  return {
    list: () => [...items],
    add(receipt) {
      items.push(receipt);
      persist();
    },
    update(id, patch) {
      const index = items.findIndex((r) => r.id === id);
      const current = items[index];
      if (index === -1 || !current) return;
      items[index] = { ...current, ...patch };
      persist();
    },
    latestFor(itemId) {
      const matching = items.filter((r) => r.itemId === itemId);
      return matching.length > 0 ? matching[matching.length - 1] ?? null : null;
    },
  };
}

  merchantUrl: string,
  settlementTx: string,

): Promise<MerchantActivity | null> {

  const target = settlementTx.toLowerCase();

    const match = items.find((a) => (a.kind === "sweep" || a.kind === "sweep_skipped") && (a.triggeredBy ?? "").toLowerCase() === target);

    if (options.nowMs() - started >= options.timeoutMs) return null;
    await options.sleep(options.pollMs);
  }

import type { TradingApi } from "@sharepay/core/binance/trading";
import {
  bscscanTx,
  initialSteps,
  type CheckoutMode,
  type CheckoutState,
  type CheckoutStep,
  type ItemId,
  type Receipt,
  type StepId,
} from "@sharepay/core/dto";
import { ONE, formatAmount, formatSignificant, mulDivUp, sharesFromTokens } from "@sharepay/core/money";
import { getStable, sameAddress } from "@sharepay/core/tokens";
import type { Address } from "viem";
import type { Holding, MarketSnapshot } from "./holdings.js";
import type { ReceiptStore } from "./ledger.js";
import { waitForReinvest } from "./merchantFeed.js";
import type { MerchantAdapter } from "./merchants/types.js";
import { estimateSale, planSale } from "./planner.js";
import type { ProbeParams, RouteOutcome } from "./routeProbe.js";
import { classifyStatus } from "./status.js";
import type { PayerWallet } from "./wallets/types.js";
import { fetchChallenge, replayWithPayment } from "./x402client.js";

export interface CheckoutDeps {
  walletFor: (itemId: ItemId) => PayerWallet;
  trading: TradingApi;
  snapshot: () => Promise<MarketSnapshot>;
  holdings: (owner: Address, snapshot: MarketSnapshot) => Promise<Holding[]>;
  probe: (params: ProbeParams) => Promise<RouteOutcome>;
  fetchImpl: typeof fetch;
  sleep: (ms: number) => Promise<void>;
  nowMs: () => number;
  receipts: ReceiptStore;
  merchantUrl: string;
  onSaleBroadcast?: () => void;
}

export interface CheckoutRequest {
  id: string;
  adapter: MerchantAdapter;
  tokenAddress: Address;
  mode: CheckoutMode;
}

export const MAX_PAYMENT_ATOMIC = 250000000000000000n;

export function formatPriceUsd(amount: bigint): string {
  const cents = mulDivUp(amount, 100n, ONE);
  return `${cents / 100n}.${(cents % 100n).toString().padStart(2, "0")}`;
}

class StepFailure extends Error {
  readonly code: string;

  constructor(code: string, message: string) {
    super(message);
    this.code = code;
  }
}

function codeOf(error: unknown): string {
  if (error instanceof StepFailure) return error.code;
  const code = (error as { code?: unknown } | null)?.code;
  return typeof code === "string" ? code : "unexpected_error";
}

export async function runCheckout(deps: CheckoutDeps, request: CheckoutRequest, emit: (state: CheckoutState) => void): Promise<CheckoutState> {
  const iso = () => new Date(deps.nowMs()).toISOString();
  const state: CheckoutState = {
    id: request.id,
    itemId: request.adapter.id,
    mode: request.mode,
    tokenAddress: request.tokenAddress,
    status: "running",
    steps: initialSteps(iso()),
    receipt: null,
    deliverable: null,
    error: null,
    etaSec: null,
    jobStartedAt: null,
    lastPolledAt: null,
    startedAt: iso(),
    finishedAt: null,
  };
  const publish = () => emit(structuredClone(state));
  const step = (id: StepId): CheckoutStep => {
    const found = state.steps.find((s) => s.id === id);
    if (!found) throw new Error(`unknown step ${id}`);
    return found;
  };
  const set = (id: StepId, patch: Partial<CheckoutStep>) => {
    Object.assign(step(id), patch, { updatedAt: iso() });
    publish();
  };
  let current: StepId = "challenge";
  const begin = (id: StepId) => {
    current = id;
    set(id, { status: "running" });
  };

  publish();
  const wallet = deps.walletFor(request.adapter.id);
  try {
    begin("challenge");
    const paid = request.adapter.request();
    const { challenge, selected } = await fetchChallenge(deps.fetchImpl, paid, {
      maxAmount: MAX_PAYMENT_ATOMIC,
      preferredAssets: [getStable("U").address, getStable("USD1").address],
    });
    const priceAtomic = BigInt(selected.accept.amount);
    const paidUsd = formatPriceUsd(priceAtomic);
    set("challenge", {
      status: "done",
      summary: `${request.adapter.title} costs $${paidUsd}`,
      detail: { merchant: request.adapter.merchantName, asset: selected.stable.symbol, amount: selected.accept.amount, payTo: selected.accept.payTo, network: selected.accept.network },
    });

    begin("quote");
    const owner = await wallet.address();
    const snapshot = await deps.snapshot();
    const holdings = await deps.holdings(owner, snapshot);
    const holding = holdings.find((h) => sameAddress(h.stock.address, request.tokenAddress));
    if (!holding) throw new StepFailure("not_held", "This wallet holds none of the selected stock");
    const status = classifyStatus(holding.status);
    if (status.cls === "hard") throw new StepFailure(status.code, status.message);
    const estimate = estimateSale({ priceAtomic, tokenPriceUsd: holding.priceUsd, sharesPerToken: holding.sharesPerToken });
    const outcome = await deps.probe({ stock: holding.stock.address, payAsset: selected.stable.address, sellAmount: estimate.sellAmount, wallet: owner });
    if (outcome.kind !== "swap") throw new StepFailure(outcome.kind, outcome.message);
    const plan = await planSale(deps.trading, { stock: holding.stock, payAsset: selected.stable.address, priceAtomic, estimate, outcome, wallet: owner, balance: holding.balance });
    const sharesSold = formatSignificant(sharesFromTokens(plan.sellAmount, holding.sharesPerToken), 18, 3);
    const tokensSold = formatSignificant(plan.sellAmount, 18, 4);
    set("quote", {
      status: "done",
      summary: `Sell ≈${sharesSold} ${holding.stock.name} shares for $${paidUsd}`,
      detail: { token: holding.stock.symbol, tokensSold, expectedOut: formatAmount(plan.expectedOut, 18, 6), legs: String(plan.legs.length), priceImpactPercent: plan.priceImpactPercent ?? "n/a", marketStatus: status.message },
    });

    begin("simulate");
    const simulation = await wallet.simulateSale(plan);
    if (!simulation.ok) throw new StepFailure("simulation_failed", simulation.failReason ?? "simulation failed");
    set("simulate", {
      status: "done",
      summary: simulation.mode === "swap" ? "The sale was simulated on BSC and succeeds" : "The approval was simulated; the swap can be simulated after it lands",
      detail: { ...simulation.detail, mode: simulation.mode, wallet: wallet.kind },
    });

    if (request.mode === "simulate") {
      for (const id of ["sell", "authorize", "settle", "deliver", "reinvest"] as const) {
        Object.assign(step(id), { status: "skipped", summary: "Simulate mode: nothing signed or broadcast", updatedAt: iso() });
      }
      state.status = "succeeded";
      state.finishedAt = iso();
      publish();
      return structuredClone(state);
    }

    begin("sell");
    // The live slot is consumed by the sale's first broadcast; a check that fails before it releases the slot.
    let slotTaken = false;
    const execution = await wallet.executeSale(plan, () => {
      if (slotTaken) return;
      slotTaken = true;
      deps.onSaleBroadcast?.();
    });
    set("sell", {
      status: "done",
      summary: `Sold ${sharesSold} ${holding.stock.name} shares`,
      detail: { received: formatAmount(execution.received, 18, 6), asset: selected.stable.symbol },
      links: execution.txHashes.map((hash, i) => ({ label: `Sale tx ${i + 1}`, href: bscscanTx(hash) })),
    });

    begin("authorize");
    const header = await wallet.signPayment(challenge, selected, Math.floor(deps.nowMs() / 1000));
    set("authorize", { status: "done", summary: `Payment of $${paidUsd} approved`, detail: { method: "EIP-3009 transferWithAuthorization", header: header.name } });

    begin("settle");
    if (request.adapter.kind === "async") {
      state.status = "awaiting_delivery";
      state.etaSec = request.adapter.etaSec;
      publish();
    }
    let replay: Awaited<ReturnType<typeof replayWithPayment>>;
    try {
      replay = await replayWithPayment(deps.fetchImpl, paid, header);
    } catch (error) {
      throw new StepFailure("delivery_unknown", `The payment was signed but the merchant did not answer: ${error instanceof Error ? error.message : String(error)}`);
    }
    const accepted = replay.response.status < 400 && (replay.settlement === null || replay.settlement.success);
    if (!accepted) {
      const reason = replay.settlement?.errorReason ?? `HTTP ${replay.response.status}`;
      throw new StepFailure("settle_failed", `The merchant refused the payment: ${reason}`);
    }
    const settlementTx = replay.settlement?.transaction ? replay.settlement.transaction : null;
    set("settle", {
      status: "done",
      summary: settlementTx ? "The merchant settled the payment on BSC" : "Settled by the merchant's facilitator",
      detail: { network: replay.settlement?.network ?? selected.accept.network, payer: replay.settlement?.payer ?? owner },
      links: settlementTx ? [{ label: "Settlement tx", href: bscscanTx(settlementTx) }] : [],
    });
    const receipt: Receipt = {
      id: request.id,
      itemId: request.adapter.id,
      itemTitle: request.adapter.title,
      merchantName: request.adapter.merchantName,
      paidUsd,
      paidAmount: formatAmount(priceAtomic, 18, 6),
      paidAsset: selected.stable.symbol,
      stockSymbol: holding.stock.symbol,
      stockName: holding.stock.name,
      sharesSold,
      tokensSold,
      swapTxHashes: execution.txHashes,
      settlementTxHash: settlementTx,
      reinvestTxHash: null,
      createdAt: iso(),
      wallet: wallet.kind,
    };
    state.receipt = receipt;
    deps.receipts.add(receipt);
    publish();

    begin("deliver");
    state.deliverable = await request.adapter.deliver(replay.response, {
      fetchImpl: deps.fetchImpl,
      sleep: deps.sleep,
      nowMs: deps.nowMs,
      onProgress: (update) => {
        state.status = "awaiting_delivery";
        state.etaSec = update.etaSec;
        state.jobStartedAt = update.jobStartedAt;
        state.lastPolledAt = update.lastPolledAt;
        set("deliver", { status: "running", summary: update.note });
      },
    });
    state.status = "running";
    state.etaSec = null;
    set("deliver", { status: "done", summary: `${request.adapter.title} delivered` });

    begin("reinvest");
    if (request.adapter.id !== "sharepay-nvda-report" || !settlementTx) {
      set("reinvest", { status: "skipped", summary: "Only SharePay's own merchant reinvests" });
    } else {
      const sweep = await waitForReinvest(deps.fetchImpl, deps.merchantUrl, settlementTx, { timeoutMs: 120_000, pollMs: 5_000, sleep: deps.sleep, nowMs: deps.nowMs });
      if (sweep && sweep.kind === "sweep") {
        const reinvestTx = sweep.txHashes[sweep.txHashes.length - 1] ?? null;
        receipt.reinvestTxHash = reinvestTx;
        deps.receipts.update(receipt.id, { reinvestTxHash: reinvestTx });
        set("reinvest", { status: "done", summary: sweep.summary, links: sweep.txHashes.map((hash, i) => ({ label: `Reinvest tx ${i + 1}`, href: bscscanTx(hash) })) });
      } else if (sweep) {
        set("reinvest", { status: "skipped", summary: sweep.summary });
      } else {
        set("reinvest", { status: "skipped", summary: "Reinvest pending on the merchant agent" });
      }
    }

    state.status = "succeeded";
    state.finishedAt = iso();
    publish();
    return structuredClone(state);
  } catch (error) {
    const message = error instanceof Error ? error.message : String(error);
    Object.assign(step(current), { status: "failed", summary: message, updatedAt: iso() });
    state.status = "failed";
    state.etaSec = null;
    state.error = { code: codeOf(error), message };
    state.finishedAt = iso();
    publish();
    return structuredClone(state);
  }
}
