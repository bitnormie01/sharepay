# Merchant Agent Implementation Plan

**Goal:** Build `@sharepay/merchant`, then mount it inside a BNB Agent Studio project
(`sharepaymerchant/`). The result is a merchant agent with these routes and behaviour:
- `GET /x402/report?symbol=NVDA` is a paid x402 v2 `exact` endpoint, 0.05 U or USD1 via EIP-3009.
- It verifies every payment and settles `transferWithAuthorization` itself, with its Studio wallet as sole signer.
- It serves a deterministic, no-LLM research report built from the RWA Data API.
- It sweeps its revenue into SPYB.

**Architecture:**
- **Pure logic.** Small modules with injected dependencies, each tested with fakes and throwaway keys: config, report, requirements, verify, settle, activity, sweeper, routes.
- **Wiring.** `runtime.ts` connects the real Binance/BSC clients.
- **Studio project.** `bag init` scaffolds it (TypeScript Express, `--llm-provider none`, bsc-mainnet). It becomes an npm workspace that imports `@sharepay/merchant`.
- **Mounting.** One import block, one replaced `runWork` line and two mount lines.

**Tech Stack:** Node.js >= 22.12, TypeScript ~5.9.3, express ^5.2.1, viem ^2.57.3, zod ^3.25.76, vitest ^3.2.7, supertest ^7.3.1, @bnbagent/studio-cli 0.0.14 (root devDependency, `npx bag`)

**Source spec:** docs/superpowers/specs/2026-10-06-sharepay-design.md

**Conventions for every task in this plan:**
- Relative imports use `.js`; workspace imports use subpaths such as `@sharepay/core/tokens`, which `@sharepay/core` exports as `"./*" -> ./src/*.ts`.
- No test touches the network or a real key.
- The gate script `npm run check -- <file>` runs whole-repo `tsc` and then the named vitest file.
- Task 8 gates on vitest alone. Type-checking the generated Studio app exhausts a 4 GB heap (its `ai`/AWS SDK types), and the merchant package it imports is already type-checked by the root tsconfig.

export interface MerchantConfig {
  publicUrl: string;

  maxTimeoutSeconds: number;
  acceptedAssets: Array<"U" | "USD1">;
  sweep: SweepConfig;
  activityPath: string | null;

  devexEventsPath: string | null;

export function findRepoRoot(start: string): string {
  let dir = resolve(start);
  for (;;) {
    const candidate = join(dir, "package.json");
    if (existsSync(candidate)) {
      try {
        const pkg = JSON.parse(readFileSync(candidate, "utf8")) as { name?: string };
        if (pkg.name === "sharepay") return dir;
      } catch {
        // unreadable package.json: keep walking up
      }
    }
    const parent = dirname(dir);
    if (parent === dir) return resolve(start);
    dir = parent;
  }
}

function bigintEnv(env: Record<string, string | undefined>, key: string, fallback: bigint): bigint {
  const raw = env[key];
  if (raw === undefined || raw === "") return fallback;
  if (!/^\d+$/.test(raw)) throw new Error(`${key} must be an integer string`);
  return BigInt(raw);
}

function numberEnv(env: Record<string, string | undefined>, key: string, fallback: number): number {
  const raw = env[key];
  if (raw === undefined || raw === "") return fallback;
  const value = Number(raw);
  if (!Number.isFinite(value)) throw new Error(`${key} must be a number`);
  return value;
}

function pathEnv(env: Record<string, string | undefined>, key: string, fallback: string, root: string): string | null {
  const raw = env[key] ?? fallback;
  if (raw === "") return null;
  return isAbsolute(raw) ? raw : join(root, raw);
}

export function loadMerchantConfig(env: Record<string, string | undefined>, cwd: string = process.cwd()): MerchantConfig {
  const publicUrl = (env.MERCHANT_PUBLIC_URL ?? "").trim().replace(/\/+$/, "");
  if (!publicUrl) throw new Error("MERCHANT_PUBLIC_URL is required");
  const root = findRepoRoot(cwd);
  const apiKey = env.BINANCE_API_KEY ?? "";
  const secretKey = env.BINANCE_API_SECRET ?? "";
  const devex = env.DEVEX_EVENTS_PATH ?? "";
  return {
    publicUrl,
    priceAtomic: bigintEnv(env, "REPORT_PRICE_ATOMIC", 50000000000000000n),
    maxTimeoutSeconds: 300,
    acceptedAssets: ["U", "USD1"],
    sweep: {
      enabled: (env.SWEEP_ENABLED ?? "true") === "true",
      minAtomic: bigintEnv(env, "SWEEP_MIN_ATOMIC", 50000000000000000n),
      floatAtomic: bigintEnv(env, "SWEEP_FLOAT_ATOMIC", 0n),
      maxAtomic: bigintEnv(env, "SWEEP_MAX_ATOMIC", 5000000000000000000n),
      maxPriceImpactPercent: numberEnv(env, "SWEEP_MAX_PRICE_IMPACT", 2),
      slippagePercent: env.SWEEP_SLIPPAGE ?? "1",
      intervalMs: numberEnv(env, "SWEEP_INTERVAL_MS", 300000),
    },
    activityPath: pathEnv(env, "ACTIVITY_PATH", "data/runtime/merchant-activity.json", root),
    activitySeedPath: pathEnv(env, "ACTIVITY_SEED_PATH", "data/merchant-activity.seed.json", root),
    reportCacheMs: 60000,
    rpcUrl: env.BSC_RPC_URL ?? DEFAULT_BSC_RPC_URL,
    binance: apiKey && secretKey ? { apiKey, secretKey } : null,
    devexEventsPath: devex ? (isAbsolute(devex) ? devex : join(root, devex)) : null,
  };
}

import { protectionLinks, type RwaApi, type StatusInfo, type UnderlyingMarket, type UnderlyingProfile } from "@sharepay/core/binance/rwa";
import type { ReportIssuer, ResearchReportDto } from "@sharepay/core/dto";
import { STOCKS, sameAddress } from "@sharepay/core/tokens";

const WEEKDAYS = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];
const PLATFORM_NAMES: Record<string, string> = { bstock: "bStocks", ondo: "Ondo" };
const OPEN_STATUSES = new Set(["premarket", "regular", "postmarket", "overnight"]);

export function formatUtcShort(ms: number): string {
  const date = new Date(ms);
  const hh = String(date.getUTCHours()).padStart(2, "0");
  const mm = String(date.getUTCMinutes()).padStart(2, "0");
  return `${WEEKDAYS[date.getUTCDay()]} ${hh}:${mm} UTC`;
}

function statusOf(status: StatusInfo | null | undefined): { open: boolean; label: string; nextOpenTime: string | null } {
  if (!status) return { open: false, label: "Status unavailable", nextOpenTime: null };
  const open = status.openState === true;
  const next = typeof status.nextOpenTime === "number" && status.nextOpenTime > 0 ? status.nextOpenTime : null;
  if (open) {
    const session = status.marketStatus && OPEN_STATUSES.has(status.marketStatus) ? status.marketStatus : status.marketStatus ?? "open";
    return { open: true, label: `Open (${session})`, nextOpenTime: null };
  }
  const why = status.reasonMsg ?? status.reasonCode ?? status.marketStatus ?? "closed";
  return {
    open: false,
    label: `Closed (${why})${next ? ` · reopens ${formatUtcShort(next)}` : ""}`,
    nextOpenTime: next ? new Date(next).toISOString() : null,
  };
}

function compact(value: string | null | undefined): string | null {
  if (value === null || value === undefined || value === "") return null;
  const n = Number(value);
  if (!Number.isFinite(n)) return value;
  if (Math.abs(n) >= 1e12) return `${(n / 1e12).toFixed(2)}T`;
  if (Math.abs(n) >= 1e9) return `${(n / 1e9).toFixed(2)}B`;
  if (Math.abs(n) >= 1e6) return `${(n / 1e6).toFixed(2)}M`;
  return n.toFixed(2);
}

function row(label: string, value: string | null | undefined): Array<{ label: string; value: string }> {
  return value === null || value === undefined || value === "" ? [] : [{ label, value }];
}

async function safe<T>(work: () => Promise<T>): Promise<T | null> {
  try {
    return await work();
  } catch {
    return null;
  }
}

export async function buildResearchReport(deps: { rwa: RwaApi; now: () => Date }, symbol: string): Promise<ResearchReportDto> {
  const ticker = symbol.toUpperCase();
  const results = (await safe(() => deps.rwa.search(ticker))) ?? [];
  const group = results.find((r) => r.ticker.toUpperCase() === ticker);
  let assets: Array<{ platformId: string; tokenSymbol: string; tokenAddress: string }> = (group?.assets ?? [])
    .filter((a) => (a.binanceChainId === "56" || a.binanceChainId === null) && (a.platformId === "bstock" || a.platformId === "ondo"))
    .map((a) => ({ platformId: a.platformId, tokenSymbol: a.tokenSymbol, tokenAddress: a.tokenContractAddress.toLowerCase() }));
  if (assets.length === 0) {
    assets = STOCKS.filter((s) => s.underlyingTicker === ticker).map((s) => ({ platformId: s.platform, tokenSymbol: s.symbol, tokenAddress: s.address }));
  }
  if (assets.length === 0) throw new Error(`unknown symbol ${ticker}`);

  const profiles: Array<UnderlyingProfile | null> = [];
  const markets: Array<UnderlyingMarket | null> = [];
  for (const asset of assets) {
    profiles.push(await safe(() => deps.rwa.profile(asset.tokenAddress)));
    markets.push(await safe(() => deps.rwa.market(asset.tokenAddress)));
  }
  const prices = (await safe(() => deps.rwa.prices(assets.map((a) => a.tokenAddress)))) ?? [];

export function createReportCache(
  build: (symbol: string) => Promise<ResearchReportDto>,
  ttlMs: number,
  nowMs: () => number,
): (symbol: string) => Promise<ResearchReportDto> {
  const entries = new Map<string, { at: number; value: Promise<ResearchReportDto> }>();
  return (symbol: string) => {
    const key = symbol.toUpperCase();
    const current = nowMs();
    const entry = entries.get(key);
    if (entry && current - entry.at < ttlMs) return entry.value;
    const value = build(key);
    entries.set(key, { at: current, value });
    value.catch(() => {
      if (entries.get(key)?.value === value) entries.delete(key);
    });
    return value;
  };
}

import { findStableByAddress, sameAddress, type StableToken } from "@sharepay/core/tokens";
import { decodeBase64Json, decodePaymentPayload } from "@sharepay/core/x402/codec";
import { domainForStable, encodeTransferWithAuthorization, recoverAuthorizationSigner } from "@sharepay/core/x402/eip3009";
import type { PaymentPayload, PaymentRequired, PaymentRequirement } from "@sharepay/core/x402/types";
import type { Address, Hex } from "viem";

export type VerifyFailure =
  | "invalid_payload"
  | "unsupported_version"
  | "requirement_mismatch"
  | "recipient_mismatch"
  | "amount_mismatch"
  | "not_yet_valid"
  | "expired"
  | "window_too_long"
  | "invalid_nonce"
  | "nonce_in_flight"
  | "invalid_signature"
  | "nonce_used"
  | "insufficient_funds"
  | "simulation_failed";

export interface VerifyDeps {
  chain: {
    authorizationState(token: Address, authorizer: Address, nonce: Hex): Promise<boolean>;
    balanceOf(token: Address, owner: Address): Promise<bigint>;
  };
  simulate(input: { from: string; to: string; value: bigint; data: string }): Promise<{ ok: boolean; status: string; failReason: string | null }>;
  settlerAddress: Address;
  nowSec: () => number;
  inFlight: Set<string>;
}

export type VerifyResult =
  | { ok: true; payload: PaymentPayload; accept: PaymentRequirement; stable: StableToken; payer: Address; calldata: Hex; inFlightKey: string }
  | { ok: false; reason: VerifyFailure; payer: string | null; detail: string };

function fail(reason: VerifyFailure, payer: string | null, detail: string): VerifyResult {
  return { ok: false, reason, payer, detail };
}

function matchesRequirement(accepted: PaymentRequirement, candidate: PaymentRequirement): boolean {
  return (
    accepted.scheme === candidate.scheme &&
    accepted.network === candidate.network &&
    sameAddress(accepted.asset, candidate.asset) &&
    accepted.amount === candidate.amount &&
    sameAddress(accepted.payTo, candidate.payTo) &&
    accepted.maxTimeoutSeconds === candidate.maxTimeoutSeconds
  );
}

export async function verifyPayment(deps: VerifyDeps, required: PaymentRequired, headerValue: string): Promise<VerifyResult> {
  let raw: unknown;
  try {
    raw = decodeBase64Json(headerValue);
  } catch {
    return fail("invalid_payload", null, "PAYMENT-SIGNATURE is not base64 JSON");
  }
  if (raw === null || typeof raw !== "object" || (raw as { x402Version?: unknown }).x402Version !== 2) {
    return fail("unsupported_version", null, "only x402Version 2 is supported");
  }
  let payload: PaymentPayload;
  try {
    payload = decodePaymentPayload(headerValue);
  } catch (error) {
    return fail("invalid_payload", null, error instanceof Error ? error.message : String(error));
  }
  const auth = payload.payload.authorization;
  const payer = auth.from;
  const accept = required.accepts.find((candidate) => matchesRequirement(payload.accepted, candidate));
  if (!accept) return fail("requirement_mismatch", payer, "accepted does not match any current requirement");
  const stable = findStableByAddress(accept.asset);
  if (!stable || stable.eip3009 === null) return fail("requirement_mismatch", payer, "asset is not an EIP-3009 stable");
  if (!sameAddress(auth.to, accept.payTo)) return fail("recipient_mismatch", payer, `authorization.to ${auth.to} != payTo ${accept.payTo}`);
  let value: bigint;
  let validAfter: bigint;
  let validBefore: bigint;
  try {
    value = BigInt(auth.value);
    validAfter = BigInt(auth.validAfter);
    validBefore = BigInt(auth.validBefore);
  } catch {
    return fail("invalid_payload", payer, "authorization numbers are not integers");
  }
  if (value !== BigInt(accept.amount)) return fail("amount_mismatch", payer, `value ${auth.value} != amount ${accept.amount}`);
  const now = BigInt(deps.nowSec());
  if (validAfter > now) return fail("not_yet_valid", payer, `validAfter ${auth.validAfter} is in the future`);
  if (validBefore < now + 6n) return fail("expired", payer, `validBefore ${auth.validBefore} has passed`);
  if (validBefore > now + 86_400n) return fail("window_too_long", payer, "validBefore is more than 24 h away");
  if (!/^0x[0-9a-fA-F]{64}$/.test(auth.nonce)) return fail("invalid_nonce", payer, "nonce must be 32 bytes of hex");
  const inFlightKey = `${stable.address}:${payer.toLowerCase()}:${auth.nonce.toLowerCase()}`;
  if (deps.inFlight.has(inFlightKey)) return fail("nonce_in_flight", payer, "this authorization is already being settled");
  const signature = payload.payload.signature as Hex;
  let recovered: Address;
  try {
    recovered = await recoverAuthorizationSigner(domainForStable(stable), auth, signature);
  } catch (error) {

  if (balance < value) return fail("insufficient_funds", payer, `balance ${balance} < ${value}`);

  const simulation = await deps.simulate({ from: deps.settlerAddress, to: stable.address, value: 0n, data: calldata });

  return { ok: true, payload, accept, stable, payer: payer as Address, calldata, inFlightKey };

import { randomUUID } from "node:crypto";
import { existsSync, mkdirSync, readFileSync, writeFileSync } from "node:fs";
import { dirname } from "node:path";
import type { MerchantActivity } from "@sharepay/core/dto";

export type NewActivity = Omit<MerchantActivity, "id" | "at">;

export interface ActivityStore {
  list(): MerchantActivity[];
  add(entry: NewActivity): MerchantActivity;
  after(id: string | null): MerchantActivity[];
}

function readList(path: string | null): MerchantActivity[] {
  if (!path || !existsSync(path)) return [];
  try {
    const parsed = JSON.parse(readFileSync(path, "utf8")) as unknown;
    return Array.isArray(parsed) ? (parsed as MerchantActivity[]) : [];
  } catch {
    return [];
  }
}

export function createActivityStore(options: {
  path: string | null;
  seedPath: string | null;
  now: () => Date;
  maxItems?: number;
}): ActivityStore {
  const maxItems = options.maxItems ?? 200;
  const seed = readList(options.seedPath);
  const seedIds = new Set(seed.map((entry) => entry.id));
  const items: MerchantActivity[] = [...seed];
  for (const entry of readList(options.path)) {
    if (!seedIds.has(entry.id) && !items.some((existing) => existing.id === entry.id)) items.push(entry);
  }
  items.sort((a, b) => a.at.localeCompare(b.at));

  function persist(): void {
    if (!options.path) return;
    try {
      mkdirSync(dirname(options.path), { recursive: true });
      writeFileSync(options.path, JSON.stringify(items.filter((entry) => !seedIds.has(entry.id)), null, 2), "utf8");
    } catch {
      // The feed is best-effort on hosts without a persistent disk.
    }
  }

  return {
    list: () => [...items],
    add(entry) {
      const created: MerchantActivity = { ...entry, id: randomUUID(), at: options.now().toISOString() };
      items.push(created);
      while (items.length > maxItems) items.shift();
      persist();
      return created;
    },
    after(id) {
      if (id === null) return [...items];
      const index = items.findIndex((entry) => entry.id === id);
      return index === -1 ? [...items] : items.slice(index + 1);
    },
  };
}

import { BinanceApiError } from "@sharepay/core/binance/client";

import type { TransactionApi } from "@sharepay/core/binance/transaction";
import type { ChainReader } from "@sharepay/core/chain/rpc";

export interface SweepDeps {
  trading: TradingApi;
  tx: TransactionApi;
  chain: ChainReader;
  signer: TxSigner;
  mutex: WalletMutex;
  activity: ActivityStore;
  config: SweepConfig;

export interface SweepState {
  lastSkipAt: Map<string, number>;

const MAX_UINT256 = 2n ** 256n - 1n;
const REROUTE_CODES = new Set([40368, 40370, 40374]);
const SKIP_DEDUPE_MS = 10 * 60 * 1000;

class SweepSkip extends Error {}

async function bestRoute(deps: SweepDeps, from: Address, to: Address, amount: bigint): Promise<QuoteRoute | null> {
  const routes = await deps.trading.quote({
    amount,
    fromTokenAddress: from,
    toTokenAddress: to,
    userWalletAddress: deps.signer.address,
    vendor: "LiquidMesh",
  });
  return bestSwapRoute(routes);
}

async function swapLeg(deps: SweepDeps, from: Address, to: Address, amount: bigint, done: Hex[]): Promise<bigint> {
  const first = await bestRoute(deps, from, to, amount);
  if (!first) throw new SweepSkip(`no SWAP route from ${from} to ${to}`);
  const impact = Number(first.priceImpactPercent ?? "0");
  if (Number.isFinite(impact) && impact > deps.config.maxPriceImpactPercent) {
    throw new SweepSkip(`price impact ${impact}% is above ${deps.config.maxPriceImpactPercent}%`);
  }
  const pipeline = { tx: deps.tx, chain: deps.chain, signer: deps.signer };
  const spender = (first.approveTarget ?? AGGREGATION_ROUTER) as Address;
  let route: QuoteRoute | null = first;
  const allowance = await deps.chain.allowance(from, deps.signer.address, spender);
  if (allowance < amount) {
    const approve = await deps.trading.approveTransaction({ tokenContractAddress: from, approveAmount: MAX_UINT256 });
    if (!sameAddress(approve.dexContractAddress, spender)) {
      throw new SweepSkip(`approve spender ${approve.dexContractAddress} does not match route spender ${spender}`);
    }
    const approval = { to: from, data: approve.data as Hex, value: 0n };
    const simulation = await simulatePrepared(deps.tx, deps.signer.address, approval);
    if (!simulation.ok) throw new SweepSkip(`approve simulation failed: ${simulation.failReason ?? simulation.status}`);
    const sent = await deps.mutex(() =>
      sendPrepared(pipeline, approval, { mevProtection: false, receiptTimeoutMs: deps.receiptTimeoutMs, gasFallback: 80_000n }),
    );
    done.push(sent.txHash);
    route = await bestRoute(deps, from, to, amount);
    if (!route) throw new SweepSkip(`no SWAP route from ${from} to ${to} after approval`);
  }
  const build = await deps.trading.swap({
    amount,
    fromTokenAddress: from,
    toTokenAddress: to,
    userWalletAddress: deps.signer.address,
    quoteId: route.quoteId,
    slippagePercent: deps.config.slippagePercent,
  });
  if ((build.executionMode ?? "").toUpperCase() !== "SWAP" || !build.tx) {
    throw new SweepSkip("swap build is not a SWAP transaction");
  }
  const swap = { to: build.tx.to as Address, data: build.tx.data as Hex, value: BigInt(build.tx.value ?? "0") };
  const simulation = await simulatePrepared(deps.tx, deps.signer.address, swap);
  if (!simulation.ok) throw new SweepSkip(`swap simulation failed: ${simulation.failReason ?? simulation.status}`);
  const before = await deps.chain.balanceOf(to, deps.signer.address);
  const sent = await deps.mutex(() =>
    sendPrepared(pipeline, swap, { mevProtection: true, receiptTimeoutMs: deps.receiptTimeoutMs, gasFallback: 400_000n }),
  );
  done.push(sent.txHash);
  const after = await deps.chain.balanceOf(to, deps.signer.address);
  return after > before ? after - before : 0n;
}

export async function runSweepOnce(
  deps: SweepDeps,
  triggeredBy: string | null,
  state: SweepState = { lastSkipAt: new Map<string, number>() },
): Promise<void> {
  const spyb = getStock("SPYB");
  const usdt = getStable("USDT");
  for (const symbol of ["U", "USD1"] as const) {
    const stable = getStable(symbol);
    const balance = await deps.chain.balanceOf(stable.address, deps.signer.address);
    const available = balance > deps.config.floatAtomic ? balance - deps.config.floatAtomic : 0n;
    const amount = available < deps.config.maxAtomic ? available : deps.config.maxAtomic;
    if (amount === 0n || amount < deps.config.minAtomic) continue;
    const amountUsd = formatAmount(amount, 18, 2);
    const done: Hex[] = [];
    try {
      let direct = true;
      try {
        if (!(await bestRoute(deps, stable.address, spyb.address, amount))) direct = false;
      } catch (error) {
        if (error instanceof BinanceApiError && error.code !== null && REROUTE_CODES.has(error.code)) direct = false;
        else throw error;
      }
      let received: bigint;
      if (direct) {
        received = await swapLeg(deps, stable.address, spyb.address, amount, done);
      } else {
        const usdtReceived = await swapLeg(deps, stable.address, usdt.address, amount, done);
        if (usdtReceived === 0n) throw new SweepSkip("first leg returned no USDT");
        received = await swapLeg(deps, usdt.address, spyb.address, usdtReceived, done);
      }
      const multiplier = await deps.chain.uiMultiplier(spyb.address);
      const shares = formatSignificant(sharesFromTokens(received, multiplier), 18, 3);
      deps.activity.add({
        kind: "sweep",
        summary: `Reinvested $${amountUsd} into ${shares} S&P 500 shares (SPYB)`,
        amountUsd,
        txHashes: done,
        payer: null,
        triggeredBy,
        spyShares: shares,
      });
    } catch (error) {
      const reason = error instanceof Error ? error.message : String(error);
      const now = deps.nowMs();
      const last = state.lastSkipAt.get(reason);
      if (last === undefined || now - last >= SKIP_DEDUPE_MS || done.length > 0) {
        state.lastSkipAt.set(reason, now);
        deps.activity.add({
          kind: "sweep_skipped",
          summary: `Reinvest postponed: ${reason}`,
          amountUsd,
          txHashes: done,
          payer: null,
          triggeredBy,
          spyShares: null,
        });
      }
    }
  }
}

export function createSweeper(deps: SweepDeps): {
  trigger(triggeredBy: string | null): void;
  start(): void;
  stop(): void;
  idle(): Promise<void>;
} {
  const state: SweepState = { lastSkipAt: new Map<string, number>() };
  let running: Promise<void> | null = null;
  let pending = false;
  let pendingTrigger: string | null = null;
  let timer: ReturnType<typeof setInterval> | null = null;

  function trigger(triggeredBy: string | null): void {
    if (!deps.config.enabled) return;
    if (running) {
      pending = true;
      pendingTrigger = triggeredBy;
      return;
    }
    running = (async () => {
      let current: string | null = triggeredBy;
      for (;;) {
        try {
          await runSweepOnce(deps, current, state);
        } catch {
          // runSweepOnce records its own skips; a failed read must not kill the loop
        }
        if (!pending) break;
        pending = false;
        current = pendingTrigger;
        pendingTrigger = null;
      }
    })().finally(() => {
      running = null;
    });
  }

  return {
    trigger,
    start() {
      if (!deps.config.enabled || timer) return;
      timer = setInterval(() => trigger(null), deps.config.intervalMs);
      timer.unref?.();
    },
    stop() {
      if (timer) clearInterval(timer);
      timer = null;
    },
    async idle() {
      while (running) await running;
    },
  };
}

import type { ResearchReportDto } from "@sharepay/core/dto";
import { formatAmount } from "@sharepay/core/money";
import { X402_NETWORK } from "@sharepay/core/tokens";
import { encodeBase64Json, encodeSettlementResponse } from "@sharepay/core/x402/codec";
import type { PaymentRequired } from "@sharepay/core/x402/types";
import express, { type Express, type Router } from "express";
import type { Address, Hex } from "viem";
import type { ActivityStore } from "./activity.js";
import type { MerchantConfig } from "./config.js";
import { buildReportRequirements } from "./requirements.js";
import type { VerifyResult } from "./verify.js";

export interface MerchantRuntime {
  config: MerchantConfig;
  payTo: Address;
  getReport: (symbol: string) => Promise<ResearchReportDto>;
  verify: (required: PaymentRequired, headerValue: string) => Promise<VerifyResult>;
  settle: (asset: Address, calldata: Hex) => Promise<{ txHash: Hex }>;
  activity: ActivityStore;
  sweeper: { trigger(triggeredBy: string | null): void; stop(): void };
  inFlight: Set<string>;
  deepHealth: () => Promise<{ rwaCode: number | null; rwaMessage: string }>;
}

function shortAddress(address: string): string {
  return address.length > 10 ? `${address.slice(0, 6)}…${address.slice(-4)}` : address;
}

function errorText(error: unknown): string {
  return error instanceof Error ? error.message : String(error);
}

export function createMerchantRouter(runtime: MerchantRuntime): Router {
  const router = express.Router();
  const { config } = runtime;
  const priceUsd = formatAmount(config.priceAtomic, 18, 2);

  function requirementsFor(symbol: string): PaymentRequired {
    return buildReportRequirements({
      priceAtomic: config.priceAtomic,
      payTo: runtime.payTo,
      resourceUrl: `${config.publicUrl}/x402/report?symbol=${symbol}`,
      maxTimeoutSeconds: config.maxTimeoutSeconds,
      assets: config.acceptedAssets,
    });
  }

  router.get("/x402/report/price", (_req, res) => {
    const required = requirementsFor("NVDA");
    res.json({ x402Version: 2, priceUsd, accepts: required.accepts });
  });

  router.get("/x402/report", async (req, res) => {
    const symbol = String(req.query.symbol ?? "NVDA").toUpperCase();
    if (!/^[A-Z]{1,8}$/.test(symbol)) {
      res.status(400).json({ error: "bad_symbol" });
      return;
    }
    const required = requirementsFor(symbol);
    const header = req.get("payment-signature");
    if (!header) {
      res.status(402).set("PAYMENT-REQUIRED", encodeBase64Json(required)).json({ error: "payment_required", paymentRequired: required });
      return;
    }
    const verified = await runtime.verify(required, header);
    if (!verified.ok) {
      res
        .status(402)
        .set("PAYMENT-REQUIRED", encodeBase64Json(required))
        .set(
          "PAYMENT-RESPONSE",
          encodeSettlementResponse({ success: false, errorReason: verified.reason, transaction: "", network: X402_NETWORK, payer: verified.payer ?? "" }),
        )
        .json({ error: verified.reason, detail: verified.detail });
      return;
    }
    runtime.inFlight.add(verified.inFlightKey);
    let txHash: Hex;
    let report: ResearchReportDto;
    try {
      try {
        report = await runtime.getReport(symbol);
      } catch (error) {
        res.status(503).json({ error: "report_unavailable", detail: errorText(error) });
        return;
      }
      try {
        txHash = (await runtime.settle(verified.stable.address, verified.calldata)).txHash;
      } catch (error) {
        res
          .status(402)
          .set(
            "PAYMENT-RESPONSE",
            encodeSettlementResponse({ success: false, errorReason: "settle_failed", transaction: "", network: X402_NETWORK, payer: verified.payer }),
          )
          .json({ error: "settle_failed", detail: errorText(error) });
        return;
      }
    } finally {
      runtime.inFlight.delete(verified.inFlightKey);
    }
    const amountUsd = formatAmount(BigInt(verified.accept.amount), 18, 2);
    runtime.activity.add({
      kind: "settlement",
      summary: `Received a $${amountUsd} payment from ${shortAddress(verified.payer)}`,
      amountUsd,
      txHashes: [txHash],
      payer: verified.payer,
      triggeredBy: null,
      spyShares: null,
    });
    runtime.sweeper.trigger(txHash);
    res
      .status(200)
      .set("PAYMENT-RESPONSE", encodeSettlementResponse({ success: true, transaction: txHash, network: X402_NETWORK, payer: verified.payer }))
      .json({ report, settlement: { transaction: txHash, network: X402_NETWORK, payer: verified.payer } });
  });

  router.get("/sharepay/activity", (req, res) => {
    const after = typeof req.query.after === "string" && req.query.after.length > 0 ? req.query.after : null;
    res.json({ items: runtime.activity.after(after) });
  });

  router.get("/sharepay/health", async (req, res) => {
    const base = { ok: true, payTo: runtime.payTo, network: X402_NETWORK, priceUsd, sweepEnabled: config.sweep.enabled };
    if (req.query.deep === "1") {
      res.json({ ...base, ...(await runtime.deepHealth()) });
      return;
    }
    res.json(base);
  });

  return router;
}

export function mountSharePayMerchant(app: Express, runtime: MerchantRuntime): void {
  app.use(createMerchantRouter(runtime));
}

import { BinanceApiError, createBinanceClient } from "@sharepay/core/binance/client";
import { createRwaApi } from "@sharepay/core/binance/rwa";
import { createTradingApi } from "@sharepay/core/binance/trading";

import { createChainReader } from "@sharepay/core/chain/rpc";

import { createActivityStore } from "./activity.js";
import { loadMerchantConfig } from "./config.js";
import { buildResearchReport, createReportCache } from "./report.js";
import type { MerchantRuntime } from "./routes.js";
import { settlePayment } from "./settle.js";
import { createSweeper } from "./sweeper.js";
import { verifyPayment } from "./verify.js";

export async function createMerchantRuntime(options: {
  env: Record<string, string | undefined>;
  signer: TxSigner;
  fetchImpl?: typeof fetch;
}): Promise<MerchantRuntime> {
  const config = loadMerchantConfig(options.env);
  if (!config.binance) {
    throw new Error("BINANCE_API_KEY and BINANCE_API_SECRET are required for the merchant agent");
  }
  const client = createBinanceClient({
    credentials: config.binance,
    fetchImpl: options.fetchImpl,
    telemetry: config.devexEventsPath ? createJsonlSink(config.devexEventsPath) : undefined,
  });
  const rwa = createRwaApi(client);
  const trading = createTradingApi(client);
  const tx = createTransactionApi(client);
  const chain = createChainReader({ rpcUrl: config.rpcUrl });
  const mutex = createWalletMutex();
  const inFlight = new Set<string>();
  const activity = createActivityStore({ path: config.activityPath, seedPath: config.activitySeedPath, now: () => new Date() });
  const getReport = createReportCache((symbol) => buildResearchReport({ rwa, now: () => new Date() }, symbol), config.reportCacheMs, Date.now);
  const sweeper = createSweeper({
    trading,
    tx,
    chain,
    signer: options.signer,
    mutex,
    activity,
    config: config.sweep,
    receiptTimeoutMs: 90_000,
    nowMs: Date.now,
  });
  sweeper.start();
  return {
    config,
    payTo: options.signer.address,
    getReport,
    verify: (required, headerValue) =>
      verifyPayment(
        {
          chain,
          simulate: (input) => tx.simulate(input),
          settlerAddress: options.signer.address,
          nowSec: () => Math.floor(Date.now() / 1000),
          inFlight,
        },
        required,
        headerValue,
      ),
    settle: (asset, calldata) => settlePayment({ pipeline: { tx, chain, signer: options.signer }, mutex, receiptTimeoutMs: 60_000 }, asset, calldata),
    activity,
    sweeper,
    inFlight,
    deepHealth: async () => {
      try {
        await rwa.tokens({ platformId: "bstock" });
        return { rwaCode: 0, rwaMessage: "ok" };
      } catch (error) {
        if (error instanceof BinanceApiError) return { rwaCode: error.code, rwaMessage: error.apiMsg };
        return { rwaCode: null, rwaMessage: error instanceof Error ? error.message : String(error) };
      }
    },
  };
}
